 let b=NSBitmapImageRep(bitmapDataPlanes:nil,pixelsWide:width,pixelsHigh:height,bitsPerSample:8,samplesPerPixel:4,hasAlpha:true,isPlanar:false,colorSpaceName:.deviceRGB,bytesPerRow:width*4,bitsPerPixel:32)!

 NSColor(calibratedRed:0.07,green:0.1,blue:0.11,alpha:1).setFill();NSRect(x:0,y:0,width:width,height:height).fill()
 var y=height-60
 for p in paths{
  (p as NSString).draw(at:NSPoint(x:20,y:y),withAttributes:[.font:NSFont.systemFont(ofSize:18),.foregroundColor:NSColor.white]);let im=NSImage(contentsOfFile:root+p)!;let h=Int(im.size.height)*scale;y-=h+15
  NSColor(calibratedWhite:0.2,alpha:1).setFill();NSRect(x:20,y:y,width:Int(im.size.width)*scale,height:h).fill();im.draw(in:NSRect(x:20,y:y,width:Int(im.size.width)*scale,height:h),from:.zero,operation:.sourceOver,fraction:1)
  y-=65
 }
 NSGraphicsContext.restoreGraphicsState();try!b.representation(using:.png,properties:[:])!.write(to:URL(fileURLWithPath:out+file))
}
