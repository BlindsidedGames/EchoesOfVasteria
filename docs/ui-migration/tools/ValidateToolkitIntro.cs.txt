using System;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Collections.Generic;
using UnityEditor;
using UnityEngine;
using UnityEngine.UIElements;
using TimelessEchoes.UI.Toolkit;
public class ValidateToolkitIntro {
 const string Key="TimelessEchoes.IntroScreenCompleted";
 const string Output="C:/Users/mattr/Documents/Repositories/EchoesOfVasteria/docs/ui-migration/validation-2026-09-27";
 ToolkitIntroScreen screen;int phase,oldPref;bool hadPref;double due,started;float oldScale;readonly List<string> results=new();
 void Check(bool b,string s)=>results.Add((b?"PASS ":"FAIL ")+s);
 public static string Main(){if(!Application.dataPath.Contains("EOV-UITK-20260927")||!Application.isPlaying)throw new Exception("Disposable Play mode required");var t=new ValidateToolkitIntro();t.oldScale=Time.timeScale;t.hadPref=PlayerPrefs.HasKey(Key);t.oldPref=PlayerPrefs.GetInt(Key);PlayerPrefs.SetInt(Key,0);TimelessEchoes.UI.TownWindowManager.Instance.CloseAllWindows();foreach(var x in Resources.FindObjectsOfTypeAll<Transform>())if(x.gameObject.scene.IsValid()&&(x.name=="Welcome"||x.name=="Dev Note"))x.gameObject.SetActive(false);var go=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/UI/Toolkit/Intro.prefab"));t.screen=go.GetComponent<ToolkitIntroScreen>();Time.timeScale=0;t.screen.Show();t.started=EditorApplication.timeSinceStartup;EditorApplication.update+=t.Tick;return "Native intro countdown validation started with timeScale=0";}
 void Tick(){if(EditorApplication.timeSinceStartup<due)return;due=EditorApplication.timeSinceStartup+1;try{switch(phase){
 case 0:Check(screen.IsVisible&&!screen.IsReady,"First-time notice starts visible with countdown");screen.TryClose();Check(screen.IsVisible&&PlayerPrefs.GetInt(Key)==0,"Countdown rejects closing and leaves seen preference unset");phase++;break;
 case 1:ScreenCapture.CaptureScreenshot(Output+"/intro-countdown.png");phase++;due=started+10.5;break;
 case 2:Check(screen.IsReady,"Ten-second countdown completes while timeScale is zero");ScreenCapture.CaptureScreenshot(Output+"/intro-ready.png");phase++;break;
 case 3:screen.TryClose();Check(!screen.IsVisible&&PlayerPrefs.GetInt(Key)==1,"Close persists existing seen key and removes native notice");Check(screen.GetComponent<UIDocument>().rootVisualElement.childCount==0,"Closed intro detaches all native content");UnityEngine.Object.Destroy(screen.gameObject);screen=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/UI/Toolkit/Intro.prefab")).GetComponent<ToolkitIntroScreen>();phase++;break;
 case 4:Check(!screen.IsVisible,"Returning player does not see notice again");Finish();break;}}
 catch(Exception ex){results.Add("FAIL "+ex);Finish();}}
 void Finish(){Time.timeScale=oldScale;if(hadPref)PlayerPrefs.SetInt(Key,oldPref);else PlayerPrefs.DeleteKey(Key);PlayerPrefs.Save();if(screen)UnityEngine.Object.Destroy(screen.gameObject);File.WriteAllLines(Output+"/intro-checks.txt",results);EditorApplication.update-=Tick;}
}
