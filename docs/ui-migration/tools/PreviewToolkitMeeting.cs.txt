using System;
using System.Linq;
using System.Reflection;
using System.Collections.Generic;
using UnityEditor;
using UnityEngine;
using UnityEngine.UIElements;
using TimelessEchoes.UI.Toolkit;
public static class PreviewToolkitMeeting {
 public static string Main(){
 if(!Application.dataPath.Contains("EOV-UITK-20260927")||!Application.isPlaying||SteamManager.Initialized||Unity.Services.Core.UnityServices.State!=Unity.Services.Core.ServicesInitializationState.Uninitialized)throw new Exception("Offline disposable Play mode required");
 foreach(var doc in UnityEngine.Object.FindObjectsByType<UIDocument>(FindObjectsInactive.Include))UnityEngine.Object.Destroy(doc.gameObject);
 TimelessEchoes.UI.TownWindowManager.Instance.CloseAllWindows();
 foreach(var t in Resources.FindObjectsOfTypeAll<Transform>())if(t.gameObject.scene.IsValid()&&(t.name=="Welcome"||t.name=="Dev Note"))t.gameObject.SetActive(false);
 var task=AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Tasks/NPC/IvanMeetingTask.prefab").GetComponent<TimelessEchoes.Tasks.TalkToNpcTask>();
 const BindingFlags F=BindingFlags.Instance|BindingFlags.NonPublic;
 var go=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/UI/Toolkit/CharacterMeeting.prefab"));var screen=go.GetComponent<ToolkitMeetingScreen>();
 screen.Init((Sprite)task.GetType().GetField("npcSprite",F).GetValue(task),(List<string>)task.GetType().GetField("lines",F).GetValue(task),()=>Debug.Log("Native meeting completion fixture: no reward callback"));
 AppDomain.CurrentDomain.SetData("UITK_MEETING_SCREEN",screen);
 return "Native Ivan meeting preview created";
 }
}
