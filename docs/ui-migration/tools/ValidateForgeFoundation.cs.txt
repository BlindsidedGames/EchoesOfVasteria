using System;using System.IO;using System.Linq;using System.Reflection;using System.Collections.Generic;using UnityEngine;using UnityEditor;using TimelessEchoes.Gear;using TimelessEchoes.Gear.UI;using TimelessEchoes.Upgrades;using Blindsided.SaveData;using Blindsided;
public static class ValidateForgeFoundation {
 const BindingFlags F=BindingFlags.Instance|BindingFlags.NonPublic;const string Out="C:/Users/mattr/Documents/Repositories/EchoesOfVasteria/docs/ui-migration/validation-2026-09-27/";
 [Serializable]public class Snapshot {public string developed,empty,missing;}
 static object Call(object o,string n,params object[] args)=>o.GetType().GetMethod(n,F).Invoke(o,args);static object Field(object o,string n)=>o.GetType().GetField(n,F).GetValue(o);
 public static string Main(){if(!Application.isPlaying||!Application.dataPath.Contains("EOV-UITK"))throw new Exception("Disposable Play required");if(!Oracle.oracle)throw new Exception("Wait for Main");CauldronManager.Instance.StopTasting();var results=new List<string>();void Check(bool b,string s)=>results.Add((b?"PASS ":"FAIL ")+s);var before=JsonUtility.FromJson<Snapshot>(File.ReadAllText("C:/Users/mattr/Documents/Repositories/EchoesOfVasteria/Library/forge-presentation-before.json"));var text=new ForgeStatisticsPresentation();Check(text.BuildStatsText(Oracle.oracle.saveData.Forge)==before.developed,"Developed Forge statistics match pre-extraction capture");Check(text.BuildStatsText(new GameData.ForgeStats())==before.empty,"Empty Forge statistics match pre-extraction capture");Check(text.BuildStatsText(null)==before.missing,"Missing Forge statistics match pre-extraction capture");var old=UnityEngine.Object.FindAnyObjectByType<ForgeWindowUI>(FindObjectsInactive.Include);var catalog=AssetDatabase.LoadAssetAtPath<ForgeCatalog>("Assets/UI/Toolkit/ForgeCatalog.asset");var session=new GameObject("Disposable Forge session").AddComponent<ForgeSession>();session.Initialize(catalog);
 var json=Type.GetType("Newtonsoft.Json.JsonConvert, Newtonsoft.Json");string backup=(string)json.GetMethod("SerializeObject",new[]{typeof(object)}).Invoke(null,new object[]{Oracle.oracle.saveData});var random=UnityEngine.Random.state;

 string Item(GearItem item)=>item==null?"null":item.slot+"|"+item.core?.name+"|"+item.rarity?.name+"|"+string.Join(";",item.affixes.Select(a=>a.stat?.name+":"+a.value.ToString("R")));

 try {var slots=(List<CoreSlotUIReferences>)Field(old,"coreSlots");Check(catalog.cores.Length==slots.Count&&catalog.cores.Select((x,i)=>x.core==slots[i].Core&&x.coreResource==slots[i].CoreResource&&x.ingotResource==slots[i].IngotResource).All(x=>x),"All imported core resource mappings match existing scene");int seed=4300;foreach(var binding in catalog.cores){foreach(string slot in EquipmentController.Instance.Slots){Restore();Call(old,"SelectCore",binding.core);old.SelectSlot(slot);UnityEngine.Random.InitState(seed);Call(old,"OnCraftClicked");var oldItem=Item((GearItem)Field(old,"lastCrafted"));var oldAmounts=Amounts();var oldEquipped=Item(EquipmentController.Instance.GetEquipped(slot));var oldCount=Oracle.oracle.saveData.Forge.TotalCrafts;Restore();session.SelectCore(binding.core);session.SelectSlot(slot);UnityEngine.Random.InitState(seed);session.Craft();Check(oldItem==Item(session.Pending)&&oldEquipped==Item(EquipmentController.Instance.GetEquipped(slot)),"Seeded craft result/equipment "+binding.core.name+" "+slot);Check(oldAmounts==Amounts()&&oldCount==Oracle.oracle.saveData.Forge.TotalCrafts,"Seeded craft resource/telemetry "+binding.core.name+" "+slot);seed++;}}
 Check(!SteamManager.Initialized&&Unity.Services.Core.UnityServices.State==Unity.Services.Core.ServicesInitializationState.Uninitialized,"Forge comparisons remain offline");}
 catch(Exception e){results.Add("FAIL "+e);}finally{session.StopAutoCrafting();Restore();UnityEngine.Random.state=random;UnityEngine.Object.DestroyImmediate(session.gameObject);}
 File.WriteAllLines(Out+"forge-foundation-checks.txt",results);return string.Join("\n",results);}
}
