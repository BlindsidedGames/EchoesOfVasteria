using System;using System.IO;using System.Linq;using System.Reflection;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.UIElements;using TimelessEchoes.UI;using TimelessEchoes.UI.Toolkit;using TimelessEchoes.Upgrades;using References.UI;
public class ValidateNativeResources {
 const string Output="C:/Users/mattr/Documents/Repositories/EchoesOfVasteria/docs/ui-migration/validation-2026-09-27";const BindingFlags F=BindingFlags.Instance|BindingFlags.NonPublic;ToolkitResourceInventoryScreen screen;ToolkitResourceInventoryDefinition definition;TownWindowManager wm;ResourceInventoryUI legacy;ResourceManager rm;List<string> checks=new();int phase;double due;double amount,received,spent;Resource target;
 VisualElement Root=>screen.GetComponent<UIDocument>().rootVisualElement;void Check(bool b,string s)=>checks.Add((b?"PASS ":"FAIL ")+s);static string Plain(string s)=>s.Replace("<b>","").Replace("</b>","");
 public static string Main(){if(!Application.isPlaying||!Application.dataPath.Contains("EOV-UITK-20260927"))throw new Exception("Disposable Play required");var t=new ValidateNativeResources{wm=TownWindowManager.Instance,screen=UnityEngine.Object.FindAnyObjectByType<ToolkitResourceInventoryScreen>(),definition=AssetDatabase.LoadAssetAtPath<ToolkitResourceInventoryDefinition>("Assets/UI/Toolkit/ResourceInventory.asset"),legacy=ResourceInventoryUI.Instance,rm=ResourceManager.Instance};if(!t.screen||!t.screen.IsOpen)throw new Exception("Open resource preview first");t.target=t.definition.resources[0];t.amount=t.rm.GetAmount(t.target);t.received=t.target.totalReceived;t.spent=t.target.totalSpent;EditorApplication.update+=t.Tick;return "Native resource inventory checks started";}
 void Tick(){if(EditorApplication.timeSinceStartup<due)return;due=EditorApplication.timeSinceStartup+.8;try{switch(phase){case 0:Check(wm.IsWindowOpen(TownWindowManager.Window.Inventory)&&screen.IsOpen,"Quest companion routes to native inventory");Check(!((GameObject)new SerializedObject(wm).FindProperty("inventory").FindPropertyRelative("window").objectReferenceValue).activeSelf,"Legacy inventory stays hidden");Check(screen.GetComponentsInChildren<UnityEngine.UI.Graphic>(true).Length==0,"Native inventory has no Canvas graphics");Compare();ScreenCapture.CaptureScreenshot(Output+"/resources-native.png");phase=10;break;
 case 10:legacy.HighlightResource(definition.resources.Last());phase=1;break;
 case 1:var last=definition.resources.Length-1;Check(Root.Q<Label>("selected-resource").text.StartsWith(definition.resources[last].name),"Existing Alter Echo highlight route selects native resource");Check(Root.Q<Button>("resource-"+definition.resources[last].name).Q(className:"eov-resource-selection").resolvedStyle.display==DisplayStyle.Flex,"Native selection visible");Check(Root.Q<ScrollView>("resource-scroll").scrollOffset.y>0,"Highlight scrolls to lower resource");ScreenCapture.CaptureScreenshot(Output+"/resources-native-selection.png");due=EditorApplication.timeSinceStartup+3.3;phase++;break;
 case 2:Check(Root.Query<VisualElement>(className:"eov-resource-selection").ToList().All(x=>x.resolvedStyle.display==DisplayStyle.None),"Timed selection expires using scaled game time");Check(Root.Q<Label>("selected-resource").text!="","Selection expiry keeps resource title like original");rm.Add(target,123,trackStats:false,eligibleForTierRoll:false);phase++;break;
 case 3:Check(Plain(Root.Q<Button>("resource-"+target.name).Q<Label>().text)==Blindsided.Utilities.CalcUtils.FormatNumber(rm.GetAmount(target),true),"Inventory event refreshes formatted amount");rm.Spend(target,rm.GetAmount(target)-amount);wm.CloseAllWindows();wm.OpenWindow(TownWindowManager.Window.Forge);phase++;break;
 case 4:Check(!screen.IsOpen,"Forge opens Info with inventory hidden");typeof(TownWindowManager).GetMethod("ToggleForgeInfo",F).Invoke(wm,null);phase++;break;
 case 5:Check(screen.IsOpen&&wm.IsWindowOpen(TownWindowManager.Window.Inventory),"Forge Inventory toggle opens native companion");ScreenCapture.CaptureScreenshot(Output+"/resources-native-forge.png");typeof(TownWindowManager).GetMethod("ToggleForgeInfo",F).Invoke(wm,null);Check(!screen.IsOpen,"Forge Info toggle closes native companion");wm.CloseAllWindows();wm.OpenWindow(TownWindowManager.Window.Inventory);phase++;break;
 case 6:Check(screen.IsOpen,"Standalone inventory route opens");for(int i=0;i<25;i++){screen.Hide();screen.Show();}Check(Root.Query<VisualElement>("resource-inventory").ToList().Count==1,"25 cycles retain one inventory tree");var d=typeof(ResourceManager).GetField("OnInventoryChanged",F).GetValue(rm) as Delegate;Check(d.GetInvocationList().Count(x=>ReferenceEquals(x.Target,screen))==1,"One inventory subscription after cycles");wm.CloseAllWindows();d=typeof(ResourceManager).GetField("OnInventoryChanged",F).GetValue(rm) as Delegate;Check(d==null||d.GetInvocationList().All(x=>!ReferenceEquals(x.Target,screen)),"Inventory subscription removed when hidden");Check(Math.Abs(rm.GetAmount(target)-amount)<.001,"Temporary resource amount restored");Check(!SteamManager.Initialized&&Unity.Services.Core.UnityServices.State==Unity.Services.Core.ServicesInitializationState.Uninitialized,"Resource tests remain offline");Finish();break;}}catch(Exception e){checks.Add("FAIL "+e);Finish();}}
 void Compare(){legacy.UpdateSlots();var so=new SerializedObject(legacy);var parent=(Transform)so.FindProperty("slotParent").objectReferenceValue;var old=parent.GetComponentsInChildren<ResourceUIReferences>(true).Where(r=>r.countText!=null&&r.transform.parent==parent&&r.gameObject.activeSelf).ToArray();Check(old.Length==definition.resources.Length,"Reference and native resource counts match ("+old.Length+")");bool counts=true,icons=true,borders=true,backgrounds=true;for(int i=0;i<Math.Min(old.Length,definition.resources.Length);i++){var b=Root.Q<Button>("resource-"+definition.resources[i].name);counts &= Plain(b.Q<Label>().text)==old[i].countText.text;icons &= b.Q<Image>().sprite==old[i].iconImage.sprite;borders &= b.style.backgroundImage.value.sprite==old[i].tierBorderImage.sprite;backgrounds &= b.Q(className:"eov-resource-slot-background").style.backgroundImage.value.sprite==old[i].newTierBackgroundImage.sprite;}Check(counts,"All resource quantities match existing formatting");Check(icons,"All unlocked/unknown icons match");Check(borders,"All tier borders match");Check(backgrounds,"All tier backgrounds match");var slots=Root.Query<Button>(className:"eov-resource-slot").ToList();Check(slots.Take(6).All(x=>Mathf.Abs(x.worldBound.y-slots[0].worldBound.y)<.01f)&&slots[6].worldBound.y>slots[0].worldBound.y,"Six columns retained");Check(Mathf.Abs(slots[0].worldBound.width-26)<.1f&&Mathf.Abs(slots[0].worldBound.height-32)<.1f,"Original 26 by 32 slot geometry retained");}
 void Finish(){EditorApplication.update-=Tick;var diff=rm.GetAmount(target)-amount;if(diff>0)rm.Spend(target,diff);else if(diff<0)rm.Add(target,-diff,trackStats:false);target.totalReceived=received;target.totalSpent=spent;wm.CloseAllWindows();wm.OpenWindow(TownWindowManager.Window.Quests);File.WriteAllLines(Output+"/resources-checks.txt",checks);}
}
