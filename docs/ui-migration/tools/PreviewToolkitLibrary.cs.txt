using System;
using System.Linq;
using System.Reflection;
using UnityEditor;
using UnityEngine;
using UnityEngine.UIElements;
using TimelessEchoes.UI;
using TimelessEchoes.UI.Toolkit;
public static class PreviewToolkitLibrary {
 public static string Main(){
  if(!Application.dataPath.Contains("EOV-UITK-20260927"))throw new Exception("Disposable project required");
  if(!Application.isPlaying)throw new Exception("Play required");
  var old=GameObject.Find("Toolkit Library Preview");if(old)UnityEngine.Object.Destroy(old);
  var manager=TownWindowManager.Instance;manager.CloseAllWindows();typeof(TownWindowManager).GetMethod("OpenWiki",BindingFlags.Instance|BindingFlags.NonPublic).Invoke(manager,null);
  foreach(var t in Resources.FindObjectsOfTypeAll<Transform>())if(t.gameObject.scene.IsValid()&&(t.name=="Welcome"||t.name=="Dev Note"))t.gameObject.SetActive(false);
  Canvas.ForceUpdateCanvases();
  var source=Resources.FindObjectsOfTypeAll<Transform>().Single(t=>t.gameObject.scene.IsValid()&&t.name=="Library_Window");
  var frame=source.Find("Image/ScrollViews") as RectTransform;
  var corners=new Vector3[4];frame.GetWorldCorners(corners);var canvas=frame.GetComponentInParent<Canvas>();
  var camera=canvas.renderMode==RenderMode.ScreenSpaceOverlay?null:canvas.worldCamera;
  var min=RectTransformUtility.WorldToScreenPoint(camera,corners[0]);var max=RectTransformUtility.WorldToScreenPoint(camera,corners[2]);
  var scale=Screen.height/432f;
  var go=new GameObject("Toolkit Library Preview");
  var settings=ToolkitPanel.CreateSettings(AssetDatabase.LoadAssetAtPath<ThemeStyleSheet>("Assets/UI/Toolkit/Runtime.tss"),AssetDatabase.LoadAssetAtPath<PanelTextSettings>("Assets/UI/Toolkit/TextSettings.asset"));settings.sortingOrder=100;
  var doc=go.AddComponent<UIDocument>();doc.panelSettings=settings;doc.rootVisualElement.pickingMode=PickingMode.Ignore;
  var view=new ToolkitBookView(AssetDatabase.LoadAssetAtPath<ToolkitBookDefinition>("Assets/UI/Toolkit/Library.asset"),AssetDatabase.LoadAssetAtPath<ToolkitTheme>("Assets/UI/Toolkit/Theme.asset"),AssetDatabase.LoadAssetAtPath<VisualTreeAsset>("Assets/UI/Toolkit/Book.uxml"));
  view.Root.style.position=Position.Absolute;view.Root.style.left=min.x/scale;view.Root.style.top=(Screen.height-max.y)/scale;view.Root.style.width=(max.x-min.x)/scale;view.Root.style.height=(max.y-min.y)/scale;
  doc.rootVisualElement.Add(view.Root);source.gameObject.SetActive(false);
  AppDomain.CurrentDomain.SetData("UITK_LIBRARY_VIEW",view);
  AppDomain.CurrentDomain.SetData("UITK_LIBRARY_SETTINGS",settings);
  return "Native Toolkit Library at "+view.Root.style.left+","+view.Root.style.top+" size "+view.Root.style.width+","+view.Root.style.height;
 }
}
