using System;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using TimelessEchoes.UI;using TimelessEchoes.UI.Toolkit;
public static class InstallNativeAlterEchoes {public static string Main(){var scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene();if(Application.isPlaying||scene.isDirty||scene.name!="Main")throw new Exception("Clean Main Edit mode required");var wm=UnityEngine.Object.FindAnyObjectByType<TownWindowManager>(FindObjectsInactive.Include);var so=new SerializedObject(wm);if(so.FindProperty("toolkitAlterEchoes").objectReferenceValue)throw new Exception("Already installed");var go=(GameObject)PrefabUtility.InstantiatePrefab(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/UI/Toolkit/AlterEchoes.prefab"),wm.transform);so.FindProperty("toolkitAlterEchoes").objectReferenceValue=go.GetComponent<ToolkitAlterEchoesScreen>();so.ApplyModifiedPropertiesWithoutUndo();EditorSceneManager.MarkSceneDirty(scene);if(!EditorSceneManager.SaveScene(scene))throw new Exception("Save failed");return "Changed existing "+GlobalObjectId.GetGlobalObjectIdSlow(wm).targetObjectId+","+GlobalObjectId.GetGlobalObjectIdSlow(wm.transform).targetObjectId;}}
