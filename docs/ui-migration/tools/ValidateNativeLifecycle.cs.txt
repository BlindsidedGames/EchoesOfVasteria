using System;using System.Linq;using System.IO;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEngine.UIElements;using UnityEditor;using TimelessEchoes;using TimelessEchoes.Hero;using TimelessEchoes.Enemies;using TimelessEchoes.Buffs;using TimelessEchoes.UI;using TimelessEchoes.UI.Toolkit;using TimelessEchoes.Stats;
public class ValidateNativeLifecycle {
 const string Out="C:/Users/mattr/Documents/Repositories/EchoesOfVasteria/docs/ui-migration/main-validation-2026-09-27/";
 List<string> results=new();int phase,index,cycle;double due;ToolkitNavigationDefinition.Entry[] maps;ToolkitRunScreen hud;int initialPanels;bool auto;BuffRecipe recipe;
 void Check(bool b,string s)=>results.Add((b?"PASS ":"FAIL ")+s);
 void Audit(string s){Check(UnityEngine.Object.FindObjectsByType<Canvas>(FindObjectsInactive.Include).All(c=>!c.enabled),s+" zero enabled Canvas");Check(UnityEngine.Object.FindObjectsByType<UnityEngine.UI.GraphicRaycaster>(FindObjectsInactive.Include).All(c=>!c.enabled),s+" zero enabled GraphicRaycaster");Check(UnityEngine.Object.FindObjectsByType<UnityEngine.UI.Graphic>(FindObjectsInactive.Include).All(c=>!c.enabled || c is TMPro.TextMeshPro),s+" zero enabled uGUI Graphic (world MeshRenderer floating text excluded)");}
 void Click(string name){var b=hud.GetComponent<UIDocument>().rootVisualElement.Q<Button>(name);if(b==null)throw new Exception("Missing native button "+name);using(var e=NavigationSubmitEvent.GetPooled()) {e.target=b;b.SendEvent(e);}}
 public static string Main(){if(!Application.isPlaying||Application.productName!="EOVUITKMainValidation")throw new Exception("Isolated Main only");var t=new ValidateNativeLifecycle();t.maps=AssetDatabase.LoadAssetAtPath<ToolkitNavigationDefinition>("Assets/UI/Toolkit/Navigation.asset").entries.Where(x=>x.action==ToolkitNavigationDefinition.Action.BeginMap).ToArray();t.hud=UnityEngine.Object.FindAnyObjectByType<ToolkitRunScreen>();t.initialPanels=Resources.FindObjectsOfTypeAll<PanelSettings>().Length;t.due=EditorApplication.timeSinceStartup+.2;EditorApplication.update+=t.Tick;return "Native lifecycle checks started, no uGUI suppression in test";}
 void Tick(){if(EditorApplication.timeSinceStartup<due)return;due=EditorApplication.timeSinceStartup+.35;try{var gm=GameManager.Instance;switch(phase){
 case 0:TownWindowManager.Instance.CloseAllWindows();Click("run-return");phase++;due+=1;break;
 case 1:Check(gm.IsInTown&&!gm.CurrentMap,"Native Return button completes town transition");Audit("Town");Click("town-joke");Check(hud.GetComponent<UIDocument>().rootVisualElement.Query<Label>().ToList().Any(l=>l.text=="Dad-o-cado"),"Native Dad-o-cado button opens joke");ScreenCapture.CaptureScreenshot(Out+"native-town-joke.png");phase++;break;
 case 2:hud.HideDetails();gm.BeginAdventure(maps[index].map);phase++;due+=1.5;break;
 case 3:if(gm.RunLoadingActive){due+=1;return;}Check(gm.CurrentMap&&!gm.IsInTown,"Run starts "+maps[index].map.name);Audit("Run "+maps[index].map.name);Check(ToolkitWorldAnchor.Active.Count>0,"Native world indicators registered "+maps[index].map.name);Click("run-breakdown");Check(hud.GetComponent<UIDocument>().rootVisualElement.Query<Label>().ToList().Any(l=>l.text=="Run Breakdown"),"Breakdown opens "+maps[index].map.name);ScreenCapture.CaptureScreenshot(Out+"native-breakdown-"+index+".png");phase++;break;
 case 4:hud.HideDetails();gm.RequestRetreat();phase++;due+=1;break;
 case 5:Check(gm.IsInTown,"Return completes "+maps[index].map.name);Check(hud.GetComponent<UIDocument>().rootVisualElement.Query<Label>().ToList().Any(l=>l.text=="Run Summary"),"Native summary displays "+maps[index].map.name);if(++index<maps.Length)phase=2;else phase++;break;
 case 6:hud.HideDetails();gm.BeginAdventure(maps[0].map);phase++;due+=2;break;
 case 7:gm.ForceHandleHeroDeath();phase++;break;
 case 8:Check(gm.DeathPromptVisible&&gm.HeroIsDead,"Native death state exposes prompt");ScreenCapture.CaptureScreenshot(Out+"native-death.png");Click("run-restart");phase++;due+=2;break;
 case 9:Check(!gm.HeroIsDead&&!gm.DeathPromptVisible&&!gm.IsInTown,"Native Restart button begins next run");Click("run-return-on-death");Check(gm.ReturnOnDeathQueued,"Native Return on death queues action");gm.ForceHandleHeroDeath();phase++;due+=1;break;
 case 10:Check(gm.IsInTown&&!gm.CurrentMap,"Queued death returns to town");gm.BeginAdventure(maps[0].map);phase++;due+=2;break;
 case 11:gm.ForceHandleHeroDeath();due+=.2;phase++;break;
 case 12:Click("run-death-return");phase++;due+=1;break;
 case 13:Check(gm.IsInTown,"Native death Return button returns to town");gm.BeginAdventure(maps[0].map);phase++;due+=2;break;
 case 14:var bm=BuffManager.Instance;bm.UnlockSlots(5);bm.UnlockAutoSlots(5);recipe=Blindsided.Utilities.AssetCache.GetAll<BuffRecipe>().First(r=>r && r.durationType!=BuffDurationType.DistancePercent && r.GetDisplayName().Contains("Experience"));bm.AssignBuff(0,recipe);bm.ClearActiveBuffs();auto=bm.IsSlotAutoCasting(0);due+=.3;phase++;break;
 case 15:Click("run-buff-0");Check(BuffManager.Instance.GetRemaining(recipe)>0,"Native buff button activates assigned recipe");var b=hud.GetComponent<UIDocument>().rootVisualElement.Q<Button>("run-buff-0");using(var e=PointerDownEvent.GetPooled(new Event{type=EventType.MouseDown,button=1})){e.target=b;b.SendEvent(e);}Check(BuffManager.Instance.IsSlotAutoCasting(0)!=auto,"Native buff right-click toggles autocast");BuffManager.Instance.ToggleSlotAutoCast(0);Time.timeScale=1;due+=4;phase++;break;
 case 16:Time.timeScale=0;Audit("Live combat/gathering");Check(GameplayStatTracker.Instance.CurrentRunDistance>0,"Actual live run advances with legacy presenters disabled");var enemy=UnityEngine.Object.FindObjectsByType<EnemyHealth>(FindObjectsInactive.Exclude).FirstOrDefault();if(enemy){enemy.SetHealthBarVisible(true);enemy.TakeDamage(1);Check(enemy.GetComponentInChildren<ToolkitWorldAnchor>(true)!=null,"Enemy health owns native world binding");}ScreenCapture.CaptureScreenshot(Out+"native-live-run.png");phase=160;break;
 case 160:hud.ShowBreakdown();phase=17;break;
 case 17:ScreenCapture.CaptureScreenshot(Out+"native-live-breakdown.png");gm.RequestRetreat();phase++;due+=2;break;
 case 18:if(!gm.IsInTown){Time.timeScale=1;due+=1;return;}Time.timeScale=0;hud.HideDetails();var screens=new[]{TownWindowManager.Window.Buffs,TownWindowManager.Window.Forge,TownWindowManager.Window.Cauldron,TownWindowManager.Window.Quests,TownWindowManager.Window.Stats,TownWindowManager.Window.Skills,TownWindowManager.Window.AlterEchoes,TownWindowManager.Window.Library,TownWindowManager.Window.Options};foreach(var screen in screens){TownWindowManager.Instance.CloseAllWindows();TownWindowManager.Instance.OpenWindow(screen);Check(TownWindowManager.Instance.IsWindowOpen(screen),"Cycle "+cycle+" "+screen);}TownWindowManager.Instance.CloseAllWindows();if(++cycle<5)return;phase++;break;
 case 19:Audit("45 repeated windows");Check(Resources.FindObjectsOfTypeAll<PanelSettings>().Length<=initialPanels+2,"PanelSettings bounded after repeated windows (before="+initialPanels+",after="+Resources.FindObjectsOfTypeAll<PanelSettings>().Length+")");Finish();break;
 }}catch(Exception ex){results.Add("FAIL "+ex);Finish();}}
 void Finish(){EditorApplication.update-=Tick;Time.timeScale=0;File.WriteAllLines(Out+"native-lifecycle.txt",results);}
}
