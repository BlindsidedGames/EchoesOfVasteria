using System;
using System.IO;
using System.Linq;
using System.Collections.Generic;
using UnityEditor;
using UnityEngine;
using UnityEngine.UIElements;
using TimelessEchoes.UI.Toolkit;
public class ValidateToolkitCredits {
 const string Output="C:/Users/mattr/Documents/Repositories/EchoesOfVasteria/docs/ui-migration/validation-2026-09-27";
 ToolkitBookView view;int phase;double due;readonly List<string> results=new();
 void Check(bool value,string name)=>results.Add((value?"PASS ":"FAIL ")+name);
 public static string Main(){if(!Application.dataPath.Contains("EOV-UITK-20260927"))throw new Exception("Disposable project required");var test=new ValidateToolkitCredits{view=(ToolkitBookView)AppDomain.CurrentDomain.GetData("UITK_LIBRARY_VIEW")};EditorApplication.update+=test.Tick;return "Credits validation started";}
 void Tick(){if(EditorApplication.timeSinceStartup<due)return;due=EditorApplication.timeSinceStartup+1;try{switch(phase++){
 case 0:var book=AssetDatabase.LoadAssetAtPath<ToolkitBookDefinition>("Assets/UI/Toolkit/Credits.asset");Check(book.sections.Length==9,"Nine active original credit entries preserved");Check(view.Root.Query<Label>().ToList().Count==18,"Every credit title and body has a native label");Check(view.Root.Query<Button>().ToList().Count==0,"Credit entries remain non-interactive");Check(Mathf.Abs(view.Scroll.contentViewport.layout.width-743)<.1f,"Credits viewport matches original 743-unit width at 3840x2160");Check(view.Scroll.verticalScroller.highValue>0,"Long credits have usable native scroll range");ScreenCapture.CaptureScreenshot(Output+"/credits-top.png");break;
 case 1:view.Scroll.scrollOffset=new Vector2(0,view.Scroll.verticalScroller.highValue);break;
 case 2:Check(view.Scroll.scrollOffset.y>0,"Credits scroll reaches final entries");Check(view.Root.Query<Label>().ToList().Any(l=>l.text.Contains("WaringMcMarrin")),"Original supporter list retained");ScreenCapture.CaptureScreenshot(Output+"/credits-bottom.png");break;
 default:var parent=view.Root.parent;var count=parent.childCount;for(int i=0;i<25;i++){var temporary=new ToolkitBookView(AssetDatabase.LoadAssetAtPath<ToolkitBookDefinition>("Assets/UI/Toolkit/Credits.asset"),AssetDatabase.LoadAssetAtPath<ToolkitTheme>("Assets/UI/Toolkit/Theme.asset"),AssetDatabase.LoadAssetAtPath<VisualTreeAsset>("Assets/UI/Toolkit/Book.uxml"));parent.Add(temporary.Root);temporary.Dispose();}Check(parent.childCount==count,"25 construct/dispose cycles leave no attached credits roots");File.WriteAllLines(Output+"/credits-checks.txt",results);EditorApplication.update-=Tick;break;}}
 catch(Exception ex){results.Add("FAIL "+ex);File.WriteAllLines(Output+"/credits-checks.txt",results);EditorApplication.update-=Tick;}}
}
