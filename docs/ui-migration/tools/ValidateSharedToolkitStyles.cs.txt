using System;using System.Linq;using System.IO;using System.Reflection;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.UIElements;using TimelessEchoes.UI;using TimelessEchoes.UI.Toolkit;
public class ValidateSharedToolkitStyles {
 const string Out="C:/Users/mattr/Documents/Repositories/EchoesOfVasteria/docs/ui-migration/validation-2026-09-27/";const string Temporary="Assets/UI/Toolkit/TemporaryTokenValidation.uss";List<string> results=new();int phase,index;double due;StyleSheet sheet;List<VisualElement> themed=new();List<(Label label,float size)> labels=new();TownWindowManager wm;TownWindowManager.Window[] windows={TownWindowManager.Window.Library,TownWindowManager.Window.Credits,TownWindowManager.Window.Buffs,TownWindowManager.Window.Skills,TownWindowManager.Window.Quests,TownWindowManager.Window.Cauldron,TownWindowManager.Window.AlterEchoes,TownWindowManager.Window.Inventory,TownWindowManager.Window.Options};
 void Check(bool b,string label)=>results.Add((b?"PASS ":"FAIL ")+label);
 public static string Main(){if(!Application.isPlaying||!Application.dataPath.Contains("EOV-UITK"))throw new Exception("Disposable Play required");var t=new ValidateSharedToolkitStyles{wm=TownWindowManager.Instance,sheet=AssetDatabase.LoadAssetAtPath<StyleSheet>(Temporary)};if(!t.wm||!t.sheet)throw new Exception("Main and temporary USS required");EditorApplication.update+=t.Tick;return "Shared theme and icon checks started";}
 List<VisualElement> Roots()=>UnityEngine.Object.FindObjectsByType<UIDocument>(FindObjectsInactive.Exclude,FindObjectsSortMode.None).Where(d=>d.rootVisualElement!=null).Select(d=>d.rootVisualElement).ToList();
 void Tick(){if(EditorApplication.timeSinceStartup<due)return;due=EditorApplication.timeSinceStartup+.6;try{switch(phase){case 0:wm.CloseAllWindows();wm.OpenWindow(TownWindowManager.Window.Library);phase++;break;
 case 1:foreach(var root in Roots()){foreach(var themeRoot in root.Query(className:"eov-theme").ToList())themed.Add(themeRoot);foreach(var label in root.Query<Label>().ToList())if(label.ClassListContains("eov-navigation-label")||label.ClassListContains("eov-chapter-title"))labels.Add((label,label.resolvedStyle.fontSize));}Check(labels.Any(l=>l.label.ClassListContains("eov-navigation-label")),"Real navigation labels found");Check(labels.Any(l=>l.label.ClassListContains("eov-chapter-title")),"Real Library headings found");foreach(var r in themed)r.styleSheets.Add(sheet);phase++;break;
 case 2:foreach(var l in labels)results.Add("METRIC "+string.Join(",",l.label.GetClasses())+" text="+l.label.text+" before="+l.size+" after="+l.label.resolvedStyle.fontSize);Check(labels.Where(l=>Math.Abs(l.size-8)<.01f).All(l=>Math.Abs(l.label.resolvedStyle.fontSize-12)<.01f),"One token changes standard headings in navigation and Library");foreach(var r in themed)r.styleSheets.Remove(sheet);phase++;break;
 case 3:Check(labels.All(l=>Math.Abs(l.label.resolvedStyle.fontSize-l.size)<.01f),"Removing override restores typography");var cache=new ToolkitWindowLayout();var element=new VisualElement();var rect=new Rect(1,2,300,200);Check(cache.Apply(element,rect),"Initial bounds applied");int writes=0;for(int i=0;i<600;i++)if(cache.Apply(element,rect))writes++;Check(writes==0,"600 unchanged layout calls cause no style writes");Check(cache.Apply(element,new Rect(1,2,301,200)),"Changed bounds applied");Check(cache.Apply(new VisualElement(),new Rect(1,2,301,200)),"New visual tree receives unchanged bounds");phase++;break;
 case 4:wm.CloseAllWindows();wm.OpenWindow(windows[index]);phase++;break;
 case 5:var images=Roots().SelectMany(r=>r.Query<Image>().ToList()).Where(i=>i.sprite).ToArray();Check(images.Length>0,"Native images found in "+windows[index]);Check(images.All(i=>i.scaleMode==ScaleMode.ScaleToFit),"All sprite images preserve aspect in "+windows[index]+" ("+images.Length+")");index++;if(index<windows.Length){phase=4;break;}wm.CloseAllWindows();wm.OpenWindow(TownWindowManager.Window.Buffs);phase++;break;
 case 6:var buffs=UnityEngine.Object.FindAnyObjectByType<ToolkitBuffsScreen>();typeof(ToolkitBuffsScreen).GetMethod("OpenPicker",BindingFlags.Instance|BindingFlags.NonPublic).Invoke(buffs,null);phase++;break;
 case 7:var picker=UnityEngine.Object.FindAnyObjectByType<ToolkitProspectorPicker>();var skills=(string[])typeof(ToolkitProspectorPicker).GetField("skills",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(picker);int wood=Array.FindIndex(skills,s=>s.IndexOf("Wood",StringComparison.OrdinalIgnoreCase)>=0);if(wood>=0)typeof(ToolkitProspectorPicker).GetMethod("Filter",BindingFlags.Instance|BindingFlags.NonPublic).Invoke(picker,new object[]{wood});phase++;break;
 case 8:var p=UnityEngine.Object.FindAnyObjectByType<ToolkitProspectorPicker>();var tall=p.GetComponent<UIDocument>().rootVisualElement.Query<Image>().ToList().Where(i=>i.sprite&&i.sprite.rect.height>i.sprite.rect.width*1.25f).ToArray();Check(tall.Length>0,"Tall tree artwork present in Prospector");Check(tall.All(i=>i.scaleMode==ScaleMode.ScaleToFit),"Tall tree artwork preserves proportions");ScreenCapture.CaptureScreenshot(Out+"shared-styles-prospector-trees.png");phase++;break;
 case 9:Check(!SteamManager.Initialized&&Unity.Services.Core.UnityServices.State==Unity.Services.Core.ServicesInitializationState.Uninitialized,"Validation remains offline");Finish();break;}}catch(Exception e){results.Add("FAIL "+e);Finish();}}
 void Finish(){EditorApplication.update-=Tick;foreach(var r in themed)r.styleSheets.Remove(sheet);var p=UnityEngine.Object.FindAnyObjectByType<ToolkitProspectorPicker>();if(p)p.Hide();wm.CloseAllWindows();File.WriteAllLines(Out+"shared-style-checks.txt",results);AssetDatabase.DeleteAsset(Temporary);}
}
