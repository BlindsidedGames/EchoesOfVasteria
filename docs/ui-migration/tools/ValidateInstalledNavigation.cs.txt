using System;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Collections.Generic;
using UnityEngine;
using UnityEditor;
using UnityEngine.UIElements;
using TimelessEchoes.UI;
using TimelessEchoes.UI.Toolkit;
public class ValidateInstalledNavigation {
 const BindingFlags F=BindingFlags.Instance|BindingFlags.Static|BindingFlags.Public|BindingFlags.NonPublic;
 const string Output="C:/Users/mattr/Documents/Repositories/EchoesOfVasteria/docs/ui-migration/validation-2026-09-27";
 ToolkitNavigationScreen screen;TownWindowManager manager;List<string> results=new();int phase,index;double due;
 int[][] dimensions={new[]{1280,720},new[]{2340,1080},new[]{3840,2160}};
 VisualElement Root=>screen.GetComponent<UIDocument>().rootVisualElement;
 void Check(bool b,string s)=>results.Add((b?"PASS ":"FAIL ")+s);
 void Submit(string id){var b=Root.Q<Button>(id);using(var e=NavigationSubmitEvent.GetPooled()){e.target=b;b.SendEvent(e);}}
 public static string Main(){if(!Application.isPlaying||!Application.dataPath.Contains("EOV-UITK-20260927"))throw new Exception("Disposable Play mode required");var test=new ValidateInstalledNavigation{manager=TownWindowManager.Instance,screen=UnityEngine.Object.FindAnyObjectByType<ToolkitNavigationScreen>()};if(!test.manager||!test.screen)throw new Exception("Wait for Main startup");foreach(var intro in UnityEngine.Object.FindObjectsByType<ToolkitIntroScreen>())intro.Hide();EditorApplication.update+=test.Tick;return "Installed navigation and desktop viewport validation started";}
 void Tick(){if(EditorApplication.timeSinceStartup<due)return;due=EditorApplication.timeSinceStartup+1;try{switch(phase){
 case 0:Check(UnityEngine.Object.FindObjectsByType<ToolkitNavigationScreen>(FindObjectsInactive.Include).Length==1,"Installed Main has exactly one native navigation host");var top=Resources.FindObjectsOfTypeAll<Transform>().Single(t=>t.gameObject.scene.IsValid()&&t.name=="TopBar");Check(!top.gameObject.activeSelf,"Original toolbar is inactive in serialized Main");Check(top.parent.Find("Toolkit Toolbar Layout Space")!=null,"Installed scene reserves original toolbar height");Check(!SteamManager.Initialized&&Unity.Services.Core.UnityServices.State==Unity.Services.Core.ServicesInitializationState.Uninitialized,"Runtime remains offline and isolated");phase++;break;
 case 1:manager.CloseAllWindows();Resize(dimensions[index][0],dimensions[index][1]);phase++;break;
 case 2:Submit("options");Check(manager.IsWindowOpen(TownWindowManager.Window.Options),"Installed native button opens Options at "+Screen.width+"x"+Screen.height);phase++;break;
 case 3:var safe=ToolkitBookScreen.CalculateSafeArea(new Vector2Int(Screen.width,Screen.height),Screen.safeArea,Blindsided.SaveData.StaticReferences.SafeAreaRatio);Check(Root.Q<Button>("options").worldBound.xMin>=safe.xMin-.2f&&Root.Q<Button>("close").worldBound.xMax<=safe.xMax+.2f,"Toolbar fits computed safe area at "+Screen.width+"x"+Screen.height);ScreenCapture.CaptureScreenshot(Output+"/navigation-installed-"+Screen.width+"-options.png");phase++;break;
 case 4:Submit("wiki");Check(manager.IsWindowOpen(TownWindowManager.Window.Library)&&!manager.IsWindowOpen(TownWindowManager.Window.Options),"Installed navigation switches Options to native Library");phase++;break;
 case 5:ScreenCapture.CaptureScreenshot(Output+"/navigation-installed-"+Screen.width+"-library.png");phase++;break;
 case 6:manager.CloseAllWindows();Submit("hub");phase++;break;
 case 7:Check(screen.OpenGroup==ToolkitNavigationDefinition.Group.Hub,"Installed Hub opens after window close");ScreenCapture.CaptureScreenshot(Output+"/navigation-installed-"+Screen.width+"-hub.png");phase++;break;
 case 8:index++;if(index<dimensions.Length){phase=1;break;}manager.CloseAllWindows();File.WriteAllLines(Output+"/installed-navigation-checks.txt",results);EditorApplication.update-=Tick;break;
 }}catch(Exception ex){results.Add("FAIL "+ex);File.WriteAllLines(Output+"/installed-navigation-checks.txt",results);EditorApplication.update-=Tick;Resize(3840,2160);}}
 static void Resize(int width,int height){
  var asm=typeof(Editor).Assembly;var sizesType=asm.GetType("UnityEditor.GameViewSizes");
  var singleton=typeof(ScriptableSingleton<>).MakeGenericType(sizesType);var sizes=singleton.GetProperty("instance",F).GetValue(null);
  var group=sizesType.GetMethod("GetGroup",F).Invoke(sizes,new object[]{0});
  var sizeType=asm.GetType("UnityEditor.GameViewSize");var enumType=asm.GetType("UnityEditor.GameViewSizeType");
  var size=Activator.CreateInstance(sizeType,F,null,new object[]{Enum.ToObject(enumType,1),width,height,"UITK baseline "+width+"x"+height},null);
  group.GetType().GetMethod("AddCustomSize",F).Invoke(group,new[]{size});
  var count=(int)group.GetType().GetMethod("GetTotalCount",F).Invoke(group,null);
  var viewType=asm.GetType("UnityEditor.GameView");var view=EditorWindow.GetWindow(viewType);
  viewType.GetProperty("selectedSizeIndex",F).SetValue(view,count-1);view.Repaint();
 }
}
