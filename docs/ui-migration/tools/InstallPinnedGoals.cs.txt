using System;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using TimelessEchoes.Quests;using TimelessEchoes.UI.Toolkit;
public static class InstallPinnedGoals {public static string Main(){var scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene();if(Application.isPlaying||scene.isDirty||scene.name!="Main")throw new Exception("Clean Main Edit mode required");var manager=UnityEngine.Object.FindAnyObjectByType<PinnedQuestUIManager>(FindObjectsInactive.Include);var so=new SerializedObject(manager);if(so.FindProperty("toolkitView").objectReferenceValue)throw new Exception("Already installed");var go=(GameObject)PrefabUtility.InstantiatePrefab(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/UI/Toolkit/PinnedGoals.prefab"),manager.transform);so.FindProperty("toolkitView").objectReferenceValue=go.GetComponent<ToolkitPinnedGoals>();so.ApplyModifiedPropertiesWithoutUndo();EditorSceneManager.MarkSceneDirty(scene);if(!EditorSceneManager.SaveScene(scene))throw new Exception("Save failed");return "Changed existing "+GlobalObjectId.GetGlobalObjectIdSlow(manager).targetObjectId+","+GlobalObjectId.GetGlobalObjectIdSlow(manager.transform).targetObjectId;}}
