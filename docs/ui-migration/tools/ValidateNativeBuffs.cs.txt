using System;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Collections.Generic;

using UnityEditor;
using UnityEngine.UIElements;

public class ValidateNativeBuffs {
 const string Output="C:/Users/mattr/Documents/Repositories/EchoesOfVasteria/docs/ui-migration/validation-2026-09-27";
 const BindingFlags F=BindingFlags.Instance|BindingFlags.NonPublic;
 ToolkitBuffsScreen screen;ToolkitProspectorPicker picker;TownWindowManager manager;BuffManager buffs;
 List<string> checks=new();int phase;double due;Dictionary<string,GameData.QuestRecord> quests;BuffRecipe[] assignments;bool[] auto;int unlocked,unlockedAuto,target;TaskData choice;
 VisualElement Root=>screen.GetComponent<UIDocument>().rootVisualElement;
 VisualElement Picker=>picker.GetComponent<UIDocument>().rootVisualElement;
 void Check(bool ok,string message)=>checks.Add((ok?"PASS ":"FAIL ")+message);
 void Submit(VisualElement root,string id){var button=root.Q<Button>(id);if(button==null)throw new Exception("Missing button "+id);using(var e=NavigationSubmitEvent.GetPooled()){e.target=button;button.SendEvent(e);}}
 public static string Main(){if(!Application.isPlaying||!Application.dataPath.Contains("EOV-UITK-20260927"))throw new Exception("Disposable Play required");var t=new ValidateNativeBuffs{manager=TownWindowManager.Instance,buffs=BuffManager.Instance};var data=Oracle.oracle.saveData;t.quests=data.Quests;t.assignments=Enumerable.Range(0,5).Select(i=>t.buffs.GetAssigned(i)).ToArray();t.auto=Enumerable.Range(0,5).Select(i=>t.buffs.IsSlotAutoCasting(i)).ToArray();t.unlocked=data.UnlockedBuffSlots;t.unlockedAuto=data.UnlockedAutoBuffSlots;t.target=data.ProspectorTaskId;foreach(var intro in UnityEngine.Object.FindObjectsByType<ToolkitIntroScreen>())intro.Hide();t.manager.CloseAllWindows();t.screen=UnityEngine.Object.FindAnyObjectByType<ToolkitBuffsScreen>();if(!t.screen)t.screen=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/UI/Toolkit/Buffs.prefab")).GetComponent<ToolkitBuffsScreen>();var so=new SerializedObject(t.manager);so.FindProperty("toolkitBuffs").objectReferenceValue=t.screen;so.ApplyModifiedPropertiesWithoutUndo();EditorApplication.update+=t.Tick;return "Native Buffs tests started; isolated synthetic gates only";}
 void Tick(){if(EditorApplication.timeSinceStartup<due)return;due=EditorApplication.timeSinceStartup+.75;try{switch(phase++){
 case 0:manager.OpenWindow(TownWindowManager.Window.Buffs);Check(screen.IsOpen&&manager.IsWindowOpen(TownWindowManager.Window.Buffs),"Native Buffs route opens");Check(screen.GetComponentsInChildren<UnityEngine.UI.Graphic>(true).Length==0,"Native Buffs prefab contains no Canvas graphics");break;
 case 1:ScreenCapture.CaptureScreenshot(Output+"/buffs-native.png");var expected=buffs.Recipes.Where(r=>r&&(!r.requiredQuest||TimelessEchoes.Quests.QuestManager.Instance.IsQuestCompleted(r.requiredQuest))).ToArray();var actual=Root.Query<VisualElement>(className:"eov-buff-row").ToList();Check(actual.Select(r=>r.name).SequenceEqual(expected.Select(r=>"recipe-"+r.name)),"Developed rows match legacy quest gates and asset order");for(int i=0;i<actual.Count;i++){var labels=actual[i].Q(className:"eov-buff-texts").Query<Label>().ToList();Check(labels[0].text==expected[i].GetDisplayName()&&labels[1].text==string.Join("\n",expected[i].GetDescriptionLines()),"Exact recipe text: "+expected[i].name);}break;
 case 2:var data=Oracle.oracle.saveData;data.UnlockedBuffSlots=5;data.UnlockedAutoBuffSlots=5;var recipe=buffs.Recipes.First(r=>Root.Q<Button>("assign-"+r.name)!=null);Submit(Root,"assign-"+recipe.name);Submit(Root,"assign-slot-0");Check(buffs.GetAssigned(0)==recipe,"Assign writes existing BuffManager slot");Submit(Root,"assign-"+recipe.name);Submit(Root,"assign-slot-1");Check(buffs.GetAssigned(1)==recipe&&buffs.GetAssigned(0)==null,"Reassignment preserves one-slot-per-recipe behavior");var before=buffs.IsSlotAutoCasting(1);Submit(Root,"assign-slot-1");Check(buffs.IsSlotAutoCasting(1)!=before,"Slot click toggles autocast outside assignment");Submit(Root,"prospector-target");picker=UnityEngine.Object.FindAnyObjectByType<ToolkitProspectorPicker>();break;
 case 3:Check(ToolkitProspectorPicker.IsOpen,"Native Prospector picker opens");var tasks=Resources.LoadAll<TaskData>("Tasks").Where(t=>t&&t.weight>0&&TaskWeightService.IsTaskUnlocked(t)).OrderBy(t=>t.taskName).ToArray();var rows=Picker.Q<ScrollView>("prospector-tasks").contentContainer.Children().ToArray();Check(rows.Select(r=>r.name).SequenceEqual(tasks.Select(t=>"task-"+t.name)),"Picker shows exactly unlocked positive-weight tasks in task-name order");choice=tasks.First(t=>t.taskIcon&&t.taskIcon.rect.height>t.taskIcon.rect.width);Submit(Picker,"task-"+choice.name);Check(buffs.ProspectorTaskId==target,"Selecting a row does not change saved target before confirmation");var icon=Picker.Q<Button>("task-"+choice.name).Q<Image>();Check(Mathf.Abs(icon.resolvedStyle.width/icon.resolvedStyle.height-choice.taskIcon.rect.width/choice.taskIcon.rect.height)<.001f,"Tall task icon preserves native aspect ratio");Check(icon.parent.style.overflow.value==Overflow.Hidden,"Task icon crops inside its frame");Submit(Picker,"cancel-target");Check(!ToolkitProspectorPicker.IsOpen&&buffs.ProspectorTaskId==target,"Cancel preserves original target");Submit(Root,"prospector-target");break;
 case 4:Submit(Picker,"task-"+choice.name);Submit(Picker,"confirm-target");Check(!ToolkitProspectorPicker.IsOpen&&buffs.ProspectorTaskId==choice.taskID,"Confirm updates existing Prospector target service");Check(Root.Q<Button>("prospector-target").Q<Label>().text.Contains(choice.taskName),"Target label refreshes after confirmation");Submit(Root,"prospector-target");Submit(Picker,"next-skill");break;
 case 5:var group=Resources.LoadAll<TaskData>("Tasks").Where(t=>t&&t.weight>0&&TaskWeightService.IsTaskUnlocked(t)).Select(t=>t.associatedSkill?t.associatedSkill.name:"Other").Distinct().OrderBy(s=>s).First();var filtered=Resources.LoadAll<TaskData>("Tasks").Where(t=>t&&t.weight>0&&TaskWeightService.IsTaskUnlocked(t)&&(t.associatedSkill?t.associatedSkill.name:"Other")==group).OrderBy(t=>t.taskName);Check(Picker.Q<ScrollView>("prospector-tasks").contentContainer.Children().Select(r=>r.name).SequenceEqual(filtered.Select(t=>"task-"+t.name)),"Skill filter matches legacy grouping");ScreenCapture.CaptureScreenshot(Output+"/prospector-native-filter.png");break;
 case 6:Submit(Picker,"previous-skill");ScreenCapture.CaptureScreenshot(Output+"/prospector-native-all.png");break;
 case 7:picker.Hide();typeof(ToolkitBuffsScreen).GetMethod("RunStarted",F).Invoke(screen,null);Check(Root.Query<Button>(className:"eov-buff-assign").ToList().All(b=>!b.enabledSelf)&&!Root.Q<Button>("prospector-target").enabledSelf,"Run start disables assignment and target changes");typeof(ToolkitBuffsScreen).GetMethod("RunEnded",F).Invoke(screen,null);Check(Root.Query<Button>(className:"eov-buff-assign").ToList().All(b=>b.enabledSelf),"Run end restores assignment");Oracle.oracle.saveData.Quests=new Dictionary<string,GameData.QuestRecord>();Oracle.oracle.saveData.UnlockedBuffSlots=1;Oracle.oracle.saveData.UnlockedAutoBuffSlots=0;screen.Hide();screen.Show();break;
 case 8:Check(Root.Query<VisualElement>(className:"eov-buff-row").ToList().Count==buffs.Recipes.Count(r=>r&&!r.requiredQuest),"Fresh quest fixture hides every gated recipe");Check(Enumerable.Range(0,5).All(i=>!Root.Q<Button>("assign-slot-"+i).enabledSelf),"No auto-slot unlock means slots cannot toggle autocast");Check(Enumerable.Range(1,4).All(i=>Root.Q<Button>("assign-slot-"+i).Q<Label>().text=="Locked"),"Locked slots retain Locked labels");var gated=buffs.Recipes.First(r=>r&&r.requiredQuest);Oracle.oracle.saveData.Quests[gated.requiredQuest.questId]=new GameData.QuestRecord{Completed=true};Blindsided.EventHandler.QuestHandin(gated.requiredQuest.questId);break;
 case 9:Check(Root.Q("recipe-"+buffs.Recipes.First(r=>r&&r.requiredQuest).name)!=null,"Quest hand-in updates open native recipe list");for(int i=0;i<25;i++){screen.Hide();screen.Show();}Check(Root.Query<VisualElement>("buffs").ToList().Count==1,"25 show/hide cycles retain one native tree");Check(UnityEngine.Object.FindObjectsByType<ToolkitProspectorPicker>().Length==1,"Repeated opens reuse one picker host");Check(!SteamManager.Initialized&&Unity.Services.Core.UnityServices.State==Unity.Services.Core.ServicesInitializationState.Uninitialized,"Runtime remains offline");manager.CloseAllWindows();Check(!screen.IsOpen&&!ToolkitProspectorPicker.IsOpen,"Global close removes screen and picker");Finish();break;
 }}catch(Exception ex){checks.Add("FAIL "+ex);Finish();}}
 void Finish(){EditorApplication.update-=Tick;var data=Oracle.oracle.saveData;data.Quests=quests;data.UnlockedBuffSlots=5;data.UnlockedAutoBuffSlots=5;for(int i=0;i<5;i++)buffs.AssignBuff(i,null);for(int i=0;i<5;i++){buffs.AssignBuff(i,assignments[i]);if(buffs.IsSlotAutoCasting(i)!=auto[i])buffs.ToggleSlotAutoCast(i);}data.UnlockedBuffSlots=unlocked;data.UnlockedAutoBuffSlots=unlockedAuto;data.ProspectorTaskId=target;screen.Hide();screen.Show();File.WriteAllLines(Output+"/buffs-checks.txt",checks);}
}
