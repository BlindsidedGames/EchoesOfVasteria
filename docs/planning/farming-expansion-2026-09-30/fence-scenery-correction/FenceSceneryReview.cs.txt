using System;
using System.IO;
using System.Linq;
using System.Collections.Generic;
using Newtonsoft.Json.Linq;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;
using UnityEngine.Tilemaps;
using Object=UnityEngine.Object;
[InitializeOnLoad]
public static class FenceSceneryReview {
 static readonly string Out=Path.GetFullPath("../Echoes Farming Planning Evidence/2026-09-30/fence-scenery-correction-2026-10-01/");
 static FenceSceneryReview(){EditorApplication.update+=Poll;}
 static void Poll(){if(EditorApplication.isCompiling||EditorApplication.isUpdating||EditorApplication.isPlayingOrWillChangePlaymode||!File.Exists(Out+"request.txt"))return;string cmd=File.ReadAllText(Out+"request.txt").Trim();File.Delete(Out+"request.txt");try{Audit(cmd);}catch(Exception ex){File.WriteAllText(Out+"error.txt",ex.ToString());}File.WriteAllText(Out+"finished.txt",DateTime.UtcNow.ToString("O"));}
 static string PathOf(Transform t)=>t.parent?PathOf(t.parent)+"/"+t.name:t.name;
 static JArray V(Vector3 v)=>new JArray(v.x,v.y,v.z);
 static bool Relevant(Vector3 v)=>v.x>=-78&&v.x<=-44&&v.y>=-39&&v.y<=12;
 static JObject Art(Sprite s,Matrix4x4 matrix,string path,int order,string layer,string kind){AssetDatabase.TryGetGUIDAndLocalFileIdentifier(s,out string guid,out long id);var m=new JArray();for(int r=0;r<4;r++)for(int c=0;c<4;c++)m.Add(matrix[r,c]);return new JObject{["path"]=path,["sprite"]=s.name,["asset"]=AssetDatabase.GetAssetPath(s),["guid"]=guid,["fileId"]=id,["rect"]=new JArray(s.rect.x,s.rect.y,s.rect.width,s.rect.height),["pivot"]=new JArray(s.pivot.x,s.pivot.y),["ppu"]=s.pixelsPerUnit,["matrix"]=m,["order"]=order,["layer"]=layer,["kind"]=kind};}
 static void Capture(Camera cam,string name,int w=1600,int h=900){var rt=new RenderTexture(w,h,24,RenderTextureFormat.ARGB32);rt.Create();var old=RenderTexture.active;try{cam.aspect=(float)w/h;cam.targetTexture=rt;RenderPipeline.SubmitRenderRequest(cam,new UniversalRenderPipeline.SingleCameraRequest{destination=rt});RenderTexture.active=rt;var t=new Texture2D(w,h,TextureFormat.RGB24,false);t.ReadPixels(new Rect(0,0,w,h),0,0);t.Apply();File.WriteAllBytes(Out+name+".png",t.EncodeToPNG());Object.DestroyImmediate(t);}finally{cam.targetTexture=null;RenderTexture.active=old;rt.Release();Object.DestroyImmediate(rt);}}
 static void Audit(string cmd){var main=SceneManager.GetActiveScene();if(main.path!="Assets/Scenes/Main.unity"||main.isDirty)throw new Exception("Main must be clean");var guards=main.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<Light2D>(true)).Select(x=>new{light=x,enabled=x.enabled,path=PathOf(x.transform)}).ToArray();var mask=EditorSceneManager.GetSceneCullingMask(main);Scene scene=default;var reports=new JArray();try{foreach(var g in guards)g.light.enabled=false;EditorSceneManager.SetSceneCullingMask(main,1UL<<40);
 var files=new[]{"Assets/Editor/FarmingPreviewsV4/A0-existing.unity","Assets/Editor/FarmingPreviewsV4/A1-first-beds.unity","Assets/Editor/FarmingPreviewsV4/A2-stump-reclamation.unity","Assets/Editor/FarmingPreviewsV4/A2b-southeast-work.unity","Assets/Editor/FarmingPreviewsV4/A3-southeast-clearing.unity","Assets/Editor/FarmingPreviewsV4/B3-terraced-rows.unity","Assets/Editor/FarmingPreviewsV4/C3-grove-courtyard.unity","Assets/Editor/FarmingOrchardRowsPreview/F-orderly-rows.unity","Assets/Editor/FarmingOrchardRowsPreview/F-all-mature-clearance.unity"};
 foreach(var file in files){scene=EditorSceneManager.OpenScene(file,OpenSceneMode.Additive);SceneManager.SetActiveScene(scene);EditorSceneManager.SetSceneCullingMask(scene,1UL<<41);var roots=scene.GetRootGameObjects();if(cmd=="fix"&&!file.Contains("A0-")&&!file.Contains("A1-")){var map=roots.SelectMany(o=>o.GetComponentsInChildren<Tilemap>(true)).Single(x=>x.name=="BG_Decor 0");foreach(var cell in new[]{new Vector3Int(13,-13,0),new Vector3Int(23,-5,0)}){if(map.GetSprite(cell)?.name!="FruitTree_Growth_YoungTree")throw new Exception("Unexpected tree at "+file+" "+cell);map.SetTile(cell,null);}EditorSceneManager.MarkSceneDirty(scene);if(!EditorSceneManager.SaveScene(scene))throw new Exception("Cannot save preview");}var art=new JArray();foreach(var sr in roots.SelectMany(o=>o.GetComponentsInChildren<SpriteRenderer>(true)).Where(x=>x.gameObject.activeInHierarchy&&x.sprite&&Relevant(x.bounds.center))){var row=Art(sr.sprite,sr.transform.localToWorldMatrix,PathOf(sr.transform),sr.sortingOrder,sr.sortingLayerName,"SpriteRenderer");row["flipX"]=sr.flipX;row["flipY"]=sr.flipY;art.Add(row);}
 foreach(var map in roots.SelectMany(o=>o.GetComponentsInChildren<Tilemap>(true)).Where(x=>x.gameObject.activeInHierarchy)){var render=map.GetComponent<TilemapRenderer>();foreach(var cell in map.cellBounds.allPositionsWithin){if(!map.HasTile(cell))continue;var s=map.GetSprite(cell);if(!s)continue;var tileTransform=map.GetTransformMatrix(cell);var position=map.CellToLocalInterpolated((Vector3)cell+map.tileAnchor);var matrix=map.transform.localToWorldMatrix*Matrix4x4.Translate(position)*tileTransform;if(!Relevant(matrix.MultiplyPoint3x4(Vector3.zero)))continue;var row=Art(s,matrix,PathOf(map.transform),render.sortingOrder,render.sortingLayerName,"Tilemap");row["cell"]=V(cell);row["worldAnchor"]=V(matrix.MultiplyPoint3x4(Vector3.zero));row["tile"]=AssetDatabase.GetAssetPath(map.GetTile(cell));row["tileColor"]=map.GetColor(cell).ToString();art.Add(row);}}
 reports.Add(new JObject{["scene"]=file,["art"]=art});if(file.Contains("F-orderly")){var cam=roots.SelectMany(o=>o.GetComponentsInChildren<Camera>(true)).First();cam.overrideSceneCullingMask=EditorSceneManager.GetSceneCullingMask(scene);cam.transform.position=new Vector3(-61.5f,-11.5f,-10);cam.orthographicSize=4;Capture(cam,cmd=="fix"?"offender-after-detail":"offender-before-detail");cam.transform.position=new Vector3(-58,-16,-10);cam.orthographicSize=9;Capture(cam,cmd=="fix"?"orchard-after-close":"orchard-before-close");cam.transform.position=new Vector3(-51.5f,-4.5f,-10);cam.orthographicSize=4;Capture(cam,cmd=="fix"?"east-edge-after-detail":"east-edge-before-detail");}if(cmd=="fix"){Directory.CreateDirectory(Out+"renders");var cam=roots.SelectMany(o=>o.GetComponentsInChildren<Camera>(true)).First();cam.overrideSceneCullingMask=EditorSceneManager.GetSceneCullingMask(scene);var stem=Path.GetFileNameWithoutExtension(file);foreach(var view in new[]{new{name="town",x=-70f,y=-10f,size=32f},new{name="normal",x=-61f,y=-11f,size=18f},new{name="north-normal",x=-67f,y=0f,size=18f},new{name="south-normal",x=-61f,y=-23f,size=18f},new{name="south-close",x=-58f,y=-27f,size=9f},new{name="north-close",x=-60f,y=-.5f,size=12f},new{name="fence-detail",x=-59f,y=5f,size=7.5f}}){cam.transform.position=new Vector3(view.x,view.y,-10);cam.orthographicSize=view.size;Capture(cam,"renders/"+stem+"-"+view.name);}if(file.Contains("F-")){cam.transform.position=new Vector3(-58,-16,-10);cam.orthographicSize=9;Capture(cam,"renders/"+stem+"-orchard-close");}}EditorSceneManager.CloseScene(scene,true);scene=default;SceneManager.SetActiveScene(main);}
 File.WriteAllText(Out+(cmd=="fix"?"scene-art-audit-after.json":"scene-art-audit.json"),reports.ToString());}finally{if(scene.IsValid())EditorSceneManager.CloseScene(scene,true);SceneManager.SetActiveScene(main);EditorSceneManager.SetSceneCullingMask(main,mask);foreach(var g in guards)g.light.enabled=g.enabled;File.WriteAllText(Out+"main-guards.json",new JObject{["mainDirtyAfterRestore"]=main.isDirty,["guards"]=new JArray(guards.Select(g=>new JObject{["path"]=g.path,["before"]=g.enabled,["after"]=g.light.enabled}))}.ToString());}}
}
