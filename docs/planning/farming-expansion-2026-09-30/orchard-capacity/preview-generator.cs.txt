using System;
using System.IO;
using System.Linq;
using Newtonsoft.Json.Linq;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;
using UnityEngine.Tilemaps;
using Object=UnityEngine.Object;

[InitializeOnLoad]
public static class OrchardCapacityPreview {
 static readonly string Out=Path.GetFullPath("../Echoes Farming Planning Evidence/2026-09-30/orchard-capacity-2026-10-01/");
 static OrchardCapacityPreview(){EditorApplication.update+=Poll;}
 static void Poll(){if(EditorApplication.isPlayingOrWillChangePlaymode||EditorApplication.isCompiling||EditorApplication.isUpdating||!File.Exists(Out+"request.txt"))return;File.Delete(Out+"request.txt");try{Build();}catch(Exception ex){File.WriteAllText(Out+"error.txt",ex.ToString());}File.WriteAllText(Out+"finished.txt",DateTime.UtcNow.ToString("O"));}
 static void Assert(bool yes,string message){if(!yes)throw new Exception(message);}
 static JArray V(Vector3 v)=>new JArray(v.x,v.y,v.z);
 static string PathOf(Transform t)=>t.parent?PathOf(t.parent)+"/"+t.name:t.name;
 static Sprite SpriteAt(string path,string name)=>AssetDatabase.LoadAllAssetsAtPath(path).OfType<Sprite>().Single(x=>x.name==name);
 static SpriteRenderer Put(Transform root,Sprite sprite,float x,float y,int order,string name){var o=new GameObject(name);o.transform.SetParent(root,false);o.transform.position=new Vector3(x,y,0);var r=o.AddComponent<SpriteRenderer>();r.sprite=sprite;r.sortingOrder=order;return r;}
 static string SoilPrefix(Func<int,int,bool> inside){string join="";if(inside(0,1))join+="N";if(inside(1,0))join+="E";if(inside(0,-1))join+="S";if(inside(-1,0))join+="W";var absent=new System.Collections.Generic.List<string>();if(!(inside(-1,0)&&inside(0,1)&&inside(-1,1)))absent.Add("NW");if(!(inside(1,0)&&inside(0,1)&&inside(1,1)))absent.Add("NE");if(!(inside(1,0)&&inside(0,-1)&&inside(1,-1)))absent.Add("SE");if(!(inside(-1,0)&&inside(0,-1)&&inside(-1,-1)))absent.Add("SW");return "Farmland_DryTanBorder_SoilJoin_"+(join==""?"None":join)+"_AbsentCorners_"+(absent.Count==0?"None":string.Join("_",absent))+"_Variant";}
 static void Capture(Camera cam,string file,int w=1600,int h=900){var rt=new RenderTexture(w,h,24,RenderTextureFormat.ARGB32);rt.Create();var previous=RenderTexture.active;try{cam.aspect=(float)w/h;cam.targetTexture=rt;RenderPipeline.SubmitRenderRequest(cam,new UniversalRenderPipeline.SingleCameraRequest{destination=rt});RenderTexture.active=rt;var t=new Texture2D(w,h,TextureFormat.RGB24,false);t.ReadPixels(new Rect(0,0,w,h),0,0);t.Apply();File.WriteAllBytes(Out+file+".png",t.EncodeToPNG());Object.DestroyImmediate(t);}finally{cam.targetTexture=null;RenderTexture.active=previous;rt.Release();Object.DestroyImmediate(rt);}}
 static System.Collections.Generic.List<Vector3Int> Cells(Tilemap map){var result=new System.Collections.Generic.List<Vector3Int>();foreach(var c in map.cellBounds.allPositionsWithin)if(map.HasTile(c))result.Add(c);return result;}
 static JObject Art(Sprite s,Matrix4x4 matrix,string path,int order,string layer,string kind){AssetDatabase.TryGetGUIDAndLocalFileIdentifier(s,out string guid,out long id);var m=new JArray();for(int r=0;r<4;r++)for(int c=0;c<4;c++)m.Add(matrix[r,c]);return new JObject{["path"]=path,["sprite"]=s.name,["asset"]=AssetDatabase.GetAssetPath(s),["guid"]=guid,["fileId"]=id,["rect"]=new JArray(s.rect.x,s.rect.y,s.rect.width,s.rect.height),["pivot"]=new JArray(s.pivot.x,s.pivot.y),["ppu"]=s.pixelsPerUnit,["matrix"]=m,["order"]=order,["layer"]=layer,["kind"]=kind};}
 static JArray AuditArt(GameObject[] roots){var art=new JArray();foreach(var r in roots.SelectMany(o=>o.GetComponentsInChildren<SpriteRenderer>(true)).Where(x=>x.gameObject.activeInHierarchy&&x.sprite)){if(r.bounds.max.x< -78||r.bounds.min.x> -44||r.bounds.max.y< -39||r.bounds.min.y>12)continue;var a=Art(r.sprite,r.transform.localToWorldMatrix,PathOf(r.transform),r.sortingOrder,r.sortingLayerName,"SpriteRenderer");a["flipX"]=r.flipX;a["flipY"]=r.flipY;art.Add(a);}foreach(var map in roots.SelectMany(o=>o.GetComponentsInChildren<Tilemap>(true)).Where(x=>x.gameObject.activeInHierarchy)){var render=map.GetComponent<TilemapRenderer>();foreach(var cell in Cells(map)){var s=map.GetSprite(cell);if(!s)continue;var matrix=map.transform.localToWorldMatrix*Matrix4x4.Translate(map.CellToLocalInterpolated((Vector3)cell+map.tileAnchor))*map.GetTransformMatrix(cell);var v=matrix.MultiplyPoint3x4(Vector3.zero);if(v.x< -78||v.x> -44||v.y< -39||v.y>12)continue;var a=Art(s,matrix,PathOf(map.transform),render.sortingOrder,render.sortingLayerName,"Tilemap");a["cell"]=V(cell);a["worldAnchor"]=V(v);a["tile"]=AssetDatabase.GetAssetPath(map.GetTile(cell));art.Add(a);}}return art;}
 static void Build(){var main=SceneManager.GetActiveScene();Assert(main.path=="Assets/Scenes/Main.unity"&&!main.isDirty,"Main must be clean Edit mode");var mask=EditorSceneManager.GetSceneCullingMask(main);var guards=main.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<Light2D>(true)).Select(x=>new{light=x,state=x.enabled,path=PathOf(x.transform)}).ToArray();Scene scene=default;var records=new JArray();var logs=new JArray();Application.LogCallback logger=(msg,stack,type)=>{if(type==LogType.Error||type==LogType.Exception||type==LogType.Warning)logs.Add(new JObject{["type"]=type.ToString(),["message"]=msg});};Application.logMessageReceived+=logger;
 try{foreach(var g in guards)g.light.enabled=false;EditorSceneManager.SetSceneCullingMask(main,1UL<<40);Directory.CreateDirectory("Assets/Editor/FarmingOrchardCapacityPreview");
 var scenarios=new[]{new{name="G6-one-tile-up",columns=3,top=-13.5f,mixed=false},new{name="H8-one-tile-up",columns=4,top=-13.5f,mixed=false},new{name="G6-clear-mature",columns=3,top=-14.125f,mixed=false},new{name="H8-clear-mature",columns=4,top=-14.125f,mixed=false},new{name="G6-clear-mixed",columns=3,top=-14.125f,mixed=true},new{name="H8-clear-mixed",columns=4,top=-14.125f,mixed=true}};
 foreach(var opt in scenarios){scene=EditorSceneManager.OpenScene("Assets/Editor/FarmingPreviewsV4/A3-southeast-clearing.unity",OpenSceneMode.Additive);SceneManager.SetActiveScene(scene);EditorSceneManager.SetSceneCullingMask(scene,1UL<<41);var originalRoots=scene.GetRootGameObjects();var before=AuditArt(originalRoots);var map=originalRoots.SelectMany(o=>o.GetComponentsInChildren<Tilemap>(true)).Single(x=>x.name=="-2 Old Path");var grid=map.layoutGrid;var delta=map.CellToWorld(new Vector3Int(0,1,0))-map.CellToWorld(Vector3Int.zero);Assert(Mathf.Abs(delta.y-1)<.0001f&&Mathf.Abs(delta.x)<.0001f,"Expected actual town tile = one world unit");
 var orchard=new GameObject("OPTIONAL Orchard Capacity - "+opt.name+" - no gameplay");SceneManager.MoveGameObjectToScene(orchard,scene);var soil=AssetDatabase.LoadAllAssetsAtPath("Assets/Art/Packs/Cute_Fantasy/Tiles/FarmLand/FarmLand_Tile.png").OfType<Sprite>().ToArray();var anchors=new JArray();float[] xs=opt.columns==3?new[]{-61f,-57f,-53f}:new[]{-62f,-58.5f,-55f,-51.5f};
 for(int row=0;row<2;row++)for(int col=0;col<opt.columns;col++){float x=xs[col],y=opt.top-row*3.5f;for(int ix=0;ix<2;ix++)for(int iy=0;iy<2;iy++){var prefix=SoilPrefix((dx,dy)=>ix+dx>=0&&ix+dx<2&&iy+dy>=0&&iy+dy<2);Put(orchard.transform,soil.First(a=>a.name.StartsWith(prefix)),x-.5f+ix,y-.5f+iy,1,"Orchard soil row"+row+" col"+col+" tile"+ix+iy);}string stage="Apple_Tree_Fruiting_WholeTree",sheet="Assets/Art/Packs/Cute_Fantasy/Crops/Apple_Tree.png";if(opt.mixed){var stages=new[]{"FruitTree_Growth_YoungTree","FruitTree_Growth_SmallSapling","FruitTree_Growth_MatureTree","Apple_Tree_Fruiting_WholeTree"};stage=stages[(row*opt.columns+col)%4];if(!stage.StartsWith("Apple"))sheet="Assets/Art/Packs/Cute_Fantasy/Crops/Fruit_Tree_Stages.png";}var sprite=SpriteAt(sheet,stage);Put(orchard.transform,sprite,x+(sprite.pivot.x-16)/16,y+(sprite.pivot.y-16)/16,3-Mathf.RoundToInt(y),"Orchard tree row"+row+" col"+col);anchors.Add(new JObject{["row"]=row,["column"]=col,["anchor"]=new JArray(x,y),["stage"]=stage});}
 Assert(JToken.DeepEquals(before,AuditArt(originalRoots)),"Source scenery/crops/paths changed");var cam=originalRoots.SelectMany(o=>o.GetComponentsInChildren<Camera>(true)).First();cam.overrideSceneCullingMask=EditorSceneManager.GetSceneCullingMask(scene);var saved="Assets/Editor/FarmingOrchardCapacityPreview/"+opt.name+".unity";EditorSceneManager.SaveScene(scene,saved);foreach(var view in new[]{new{name="close",x=-57.5f,y=-16f,size=9f},new{name="normal",x=-61f,y=-17f,size=18f},new{name="town",x=-70f,y=-10f,size=32f},new{name="fence-detail",x=-57.5f,y=-11.5f,size=4f}}){cam.transform.position=new Vector3(view.x,view.y,-10);cam.orthographicSize=view.size;Capture(cam,opt.name+"-"+view.name);}
 records.Add(new JObject{["scene"]=saved,["columns"]=opt.columns,["rows"]=2,["treeCount"]=2*opt.columns,["topY"]=opt.top,["originalTopY"]=-14.5,["oneTileRequestedTopY"]=-13.5,["actualGridCellSize"]=V(grid.cellSize),["actualGridStepUp"]=V(delta),["rowSpacing"]=3.5,["columnSpacing"]=xs[1]-xs[0],["anchors"]=anchors,["originalArt"]=before,["art"]=AuditArt(scene.GetRootGameObjects()),["allOriginalArtUnchanged"]=true,["pathTilesUnchanged"]=true,["cropRendererCount"]=originalRoots.SelectMany(o=>o.GetComponentsInChildren<SpriteRenderer>(true)).Count(r=>r.name.StartsWith("Crop "))});EditorSceneManager.CloseScene(scene,true);scene=default;SceneManager.SetActiveScene(main);}
 File.WriteAllText(Out+"scene-art-audit.json",records.ToString());File.WriteAllText(Out+"render-logs.json",logs.ToString());}
 finally{if(scene.IsValid())EditorSceneManager.CloseScene(scene,true);SceneManager.SetActiveScene(main);EditorSceneManager.SetSceneCullingMask(main,mask);foreach(var g in guards)g.light.enabled=g.state;Application.logMessageReceived-=logger;File.WriteAllText(Out+"main-guards.json",new JObject{["mainDirtyAfterRestore"]=main.isDirty,["guards"]=new JArray(guards.Select(g=>new JObject{["path"]=g.path,["before"]=g.state,["after"]=g.light.enabled}))}.ToString());}}
}
