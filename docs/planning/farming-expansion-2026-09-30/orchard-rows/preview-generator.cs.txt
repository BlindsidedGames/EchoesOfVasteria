using System;
using System.IO;
using System.Linq;
using Newtonsoft.Json.Linq;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;
using UnityEngine.Tilemaps;
using Object=UnityEngine.Object;

[InitializeOnLoad]
public static class OrchardRowsPreview {
 static readonly string Out=Path.GetFullPath("../Echoes Farming Planning Evidence/2026-09-30/orchard-rows-2026-10-01/");
 static OrchardRowsPreview(){EditorApplication.update+=Poll;}
 static void Poll(){if(EditorApplication.isPlayingOrWillChangePlaymode||EditorApplication.isCompiling||EditorApplication.isUpdating||!File.Exists(Out+"request.txt"))return;File.Delete(Out+"request.txt");try{Build();}catch(Exception ex){File.WriteAllText(Out+"error.txt",ex.ToString());}File.WriteAllText(Out+"finished.txt",DateTime.UtcNow.ToString("O"));}
 static void Assert(bool yes,string message){if(!yes)throw new Exception(message);}
 static JArray V(Vector3 v)=>new JArray(v.x,v.y,v.z);
 static string PathOf(Transform t)=>t.parent?PathOf(t.parent)+"/"+t.name:t.name;
 static Sprite SpriteAt(string path,string name)=>AssetDatabase.LoadAllAssetsAtPath(path).OfType<Sprite>().Single(x=>x.name==name);
 static SpriteRenderer Put(Transform root,Sprite sprite,float x,float y,int order,string name){var o=new GameObject(name);o.transform.SetParent(root,false);o.transform.position=new Vector3(x,y,0);var r=o.AddComponent<SpriteRenderer>();r.sprite=sprite;r.sortingOrder=order;return r;}
 static string SoilPrefix(Func<int,int,bool> inside){string join="";if(inside(0,1))join+="N";if(inside(1,0))join+="E";if(inside(0,-1))join+="S";if(inside(-1,0))join+="W";var absent=new System.Collections.Generic.List<string>();if(!(inside(-1,0)&&inside(0,1)&&inside(-1,1)))absent.Add("NW");if(!(inside(1,0)&&inside(0,1)&&inside(1,1)))absent.Add("NE");if(!(inside(1,0)&&inside(0,-1)&&inside(1,-1)))absent.Add("SE");if(!(inside(-1,0)&&inside(0,-1)&&inside(-1,-1)))absent.Add("SW");return "Farmland_DryTanBorder_SoilJoin_"+(join==""?"None":join)+"_AbsentCorners_"+(absent.Count==0?"None":string.Join("_",absent))+"_Variant";}
 static void Capture(Camera cam,string file,int w=1600,int h=900){var rt=new RenderTexture(w,h,24,RenderTextureFormat.ARGB32);rt.Create();var previous=RenderTexture.active;try{cam.aspect=(float)w/h;cam.targetTexture=rt;RenderPipeline.SubmitRenderRequest(cam,new UniversalRenderPipeline.SingleCameraRequest{destination=rt});RenderTexture.active=rt;var t=new Texture2D(w,h,TextureFormat.RGB24,false);t.ReadPixels(new Rect(0,0,w,h),0,0);t.Apply();File.WriteAllBytes(Out+file+".png",t.EncodeToPNG());Object.DestroyImmediate(t);}finally{cam.targetTexture=null;RenderTexture.active=previous;rt.Release();Object.DestroyImmediate(rt);}}
 static System.Collections.Generic.List<Vector3Int> Cells(Tilemap map){var result=new System.Collections.Generic.List<Vector3Int>();foreach(var c in map.cellBounds.allPositionsWithin)if(map.HasTile(c))result.Add(c);return result;}
 static void Build(){var main=SceneManager.GetActiveScene();Assert(main.path=="Assets/Scenes/Main.unity"&&!main.isDirty,"Main must be clean Edit mode");var oldMask=EditorSceneManager.GetSceneCullingMask(main);EditorSceneManager.SetSceneCullingMask(main,1UL<<40);var guards=main.GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<Light2D>(true)).Select(x=>new{light=x,state=x.enabled,path=PathOf(x.transform)}).ToArray();Scene preview=default;var logs=new JArray();Application.LogCallback log=(message,stack,type)=>{if(type==LogType.Error||type==LogType.Exception||type==LogType.Warning)logs.Add(new JObject{["type"]=type.ToString(),["message"]=message});};Application.logMessageReceived+=log;
 try{foreach(var g in guards)g.light.enabled=false;preview=EditorSceneManager.OpenScene("Assets/Editor/FarmingPreviewsV4/A3-southeast-clearing.unity",OpenSceneMode.Additive);SceneManager.SetActiveScene(preview);EditorSceneManager.SetSceneCullingMask(preview,1UL<<41);var roots=preview.GetRootGameObjects();var sourceSprites=roots.SelectMany(x=>x.GetComponentsInChildren<SpriteRenderer>(true)).ToArray();var sourcePoses=new JArray(sourceSprites.Select(x=>new JObject{["path"]=PathOf(x.transform),["sprite"]=x.sprite?x.sprite.name:"",["position"]=V(x.transform.position)}));var paths=roots.SelectMany(x=>x.GetComponentsInChildren<Tilemap>(true)).Single(x=>x.name=="-2 Old Path");var oldTiles=Cells(paths).Select(c=>new{cell=c,tile=paths.GetTile(c)}).ToArray();var cam=roots.SelectMany(x=>x.GetComponentsInChildren<Camera>(true)).First();cam.overrideSceneCullingMask=EditorSceneManager.GetSceneCullingMask(preview);cam.transform.position=new Vector3(-58,-16,-10);cam.orthographicSize=9;Capture(cam,"baseline-close");
 var root=new GameObject("OPTIONAL Separate Orchard - two orderly rows, dedicated tree soil, no gameplay");SceneManager.MoveGameObjectToScene(root,preview);var soilSprites=AssetDatabase.LoadAllAssetsAtPath("Assets/Art/Packs/Cute_Fantasy/Tiles/FarmLand/FarmLand_Tile.png").OfType<Sprite>().ToArray();var plots=new JArray();var tiles=new JArray();var trees=new System.Collections.Generic.List<SpriteRenderer>();string stageSheet="Assets/Art/Packs/Cute_Fantasy/Crops/Fruit_Tree_Stages.png",appleSheet="Assets/Art/Packs/Cute_Fantasy/Crops/Apple_Tree.png";
 var specs=new[]{new{x=-61f,y=-14.5f,name="FruitTree_Growth_YoungTree",path=stageSheet},new{x=-56f,y=-14.5f,name="Apple_Tree_Fruiting_WholeTree",path=appleSheet},new{x=-61f,y=-18f,name="FruitTree_Growth_SmallSapling",path=stageSheet},new{x=-56f,y=-18f,name="FruitTree_Growth_MatureTree",path=stageSheet}};
 foreach(var spec in specs){var bounds=new Bounds(new Vector3(spec.x,spec.y,0),new Vector3(2,2,.2f));var blockers=sourceSprites.Where(r=>r.gameObject.activeInHierarchy&&r.sprite&&r.bounds.Intersects(bounds)).Select(r=>PathOf(r.transform)).ToArray();Assert(blockers.Length==0,"Plot intersects existing scenery: "+string.Join(",",blockers));var overlaps=oldTiles.Where(item=>{var b=new Bounds(paths.CellToWorld(item.cell)+new Vector3(.5f,.5f,0),new Vector3(.999f,.999f,.2f));return b.Intersects(bounds);}).Select(item=>V(paths.GetCellCenterWorld(item.cell))).ToArray();Assert(overlaps.Length==0,"Soil overlaps existing path");
 for(int x=0;x<2;x++)for(int y=0;y<2;y++){var prefix=SoilPrefix((dx,dy)=>x+dx>=0&&x+dx<2&&y+dy>=0&&y+dy<2);var sprite=soilSprites.First(s=>s.name.StartsWith(prefix));Put(root.transform,sprite,spec.x-.5f+x,spec.y-.5f+y,1,"Orchard-specific soil with matching edge/corner");tiles.Add(new JObject{["treeAnchor"]=new JArray(spec.x,spec.y),["position"]=new JArray(spec.x-.5f+x,spec.y-.5f+y),["sprite"]=sprite.name});}
 var art=SpriteAt(spec.path,spec.name);var tree=Put(root.transform,art,spec.x+(art.pivot.x-16)/16,spec.y+(art.pivot.y-16)/16,3-Mathf.RoundToInt(spec.y),"Optional orderly orchard tree - "+spec.name);trees.Add(tree);plots.Add(new JObject{["anchor"]=new JArray(spec.x,spec.y),["soilBoundsMin"]=V(bounds.min),["soilBoundsMax"]=V(bounds.max),["stage"]=spec.name,["spriteCanvasMin"]=V(tree.bounds.min),["spriteCanvasMax"]=V(tree.bounds.max)});
 }
 var after=new JArray(sourceSprites.Select(x=>new JObject{["path"]=PathOf(x.transform),["sprite"]=x.sprite?x.sprite.name:"",["position"]=V(x.transform.position)}));Assert(JToken.DeepEquals(sourcePoses,after),"Original sprites changed");Assert(oldTiles.All(x=>paths.GetTile(x.cell)==x.tile),"Existing path tiles changed");
 // Central 3-unit grass aisle, reachable from logging lane without covering any standing scenery.
 var aisle=new Bounds(new Vector3(-58.5f,-17.5f,0),new Vector3(1.5f,7,.2f));var aisleBlockers=sourceSprites.Where(r=>r.gameObject.activeInHierarchy&&r.sprite&&r.bounds.Intersects(aisle)).Select(r=>new JObject{["path"]=PathOf(r.transform),["boundsMin"]=V(r.bounds.min),["boundsMax"]=V(r.bounds.max)}).ToArray();Assert(aisleBlockers.Length==0,"Central aisle contains scenery");var access=oldTiles.Where(x=>{var b=new Bounds(paths.CellToWorld(x.cell)+new Vector3(.5f,.5f,0),new Vector3(1,1,.2f));return b.Intersects(aisle);}).Select(x=>V(paths.GetCellCenterWorld(x.cell))).ToArray();Assert(access.Length>0,"Central aisle does not meet existing logging path");
 Capture(cam,"rows-mixed-close");cam.transform.position=new Vector3(-61,-17,-10);cam.orthographicSize=18;Capture(cam,"rows-mixed-normal");cam.transform.position=new Vector3(-70,-10,-10);cam.orthographicSize=32;Capture(cam,"rows-town-context");
 Directory.CreateDirectory("Assets/Editor/FarmingOrchardRowsPreview");EditorSceneManager.SaveScene(preview,"Assets/Editor/FarmingOrchardRowsPreview/F-orderly-rows.unity");var mature=SpriteAt(appleSheet,"Apple_Tree_Fruiting_WholeTree");for(int i=0;i<trees.Count;i++){trees[i].sprite=mature;trees[i].transform.position=new Vector3(specs[i].x+(mature.pivot.x-16)/16,specs[i].y+(mature.pivot.y-16)/16,0);}cam.transform.position=new Vector3(-58,-16,-10);cam.orthographicSize=9;Capture(cam,"rows-all-mature-close");EditorSceneManager.SaveScene(preview,"Assets/Editor/FarmingOrchardRowsPreview/F-all-mature-clearance.unity");
 File.WriteAllText(Out+"geometry-proof.json",new JObject{["conceptual"]=true,["rows"]=2,["columns"]=2,["treeCount"]=4,["horizontalSpacing"]=5,["verticalSpacing"]=3.5,["plantingPlotSize"]=new JArray(2,2),["soilTiles"]=tiles,["plots"]=plots,["columnGrassAisleWidth"]=3,["soilRowGrassGap"]=1.5,["matureVisibleRowClearance"]=.625,["matureVisibleColumnClearance"]=3.3125,["matureVisibleClearanceFromMiddleFieldSouthFence"]=.4375,["existingPathTileCountPreserved"]=oldTiles.Length,["existingPathAisleAccessCells"]=new JArray(access),["verifiedEntranceCorridorWidth"]=1.5,["aisleBlockers"]=new JArray(aisleBlockers),["cropRenderersPreserved"]=sourceSprites.Count(x=>x.name.StartsWith("Crop ")),["allOriginalSpritePosesPreserved"]=true,["sixCropBedsRetained"]=true,["newPathCells"]=0,["newStandingTreeRemoval"]=0,["addedOrchardSoilTiles"]=16,["runtimeNavigationTested"]=false,["successfulRenderLogs"]=logs,["mainDirty"]=main.isDirty}.ToString());
 }finally{if(preview.IsValid())EditorSceneManager.CloseScene(preview,true);SceneManager.SetActiveScene(main);EditorSceneManager.SetSceneCullingMask(main,oldMask);foreach(var g in guards)g.light.enabled=g.state;Application.logMessageReceived-=log;File.WriteAllText(Out+"main-guard-proof.json",new JObject{["mainDirtyAfterRestore"]=main.isDirty,["guards"]=new JArray(guards.Select(g=>new JObject{["path"]=g.path,["before"]=g.state,["after"]=g.light.enabled}))}.ToString());}}
}
