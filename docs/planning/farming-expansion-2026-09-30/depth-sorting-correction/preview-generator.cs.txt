[InitializeOnLoad]
public static class DepthSortingPreview {
 static readonly string Out=Path.GetFullPath("../Echoes Farming Planning Evidence/2026-09-30/depth-sorting-correction-2026-10-01/");
 static DepthSortingPreview(){EditorApplication.update+=Poll;}
 static void Poll(){if(EditorApplication.isPlayingOrWillChangePlaymode||EditorApplication.isCompiling||EditorApplication.isUpdating||!File.Exists(Out+"request.txt"))return;var command=File.ReadAllText(Out+"request.txt").Trim();File.Delete(Out+"request.txt");try{if(command=="fixture")Fixture();else if(command=="recapture")Recapture();else if(command=="orchard-control")OrchardControl();else BuildAll();}catch(Exception ex){File.WriteAllText(Out+"error.txt",ex.ToString());}File.WriteAllText(Out+"finished.txt",DateTime.UtcNow.ToString("O"));}
 static void Assert(bool yes,string message){if(!yes)throw new Exception(message);}
 static JArray V(Vector3 v)=>new JArray(v.x,v.y,v.z);
 static string PathOf(Transform t)=>t.parent?PathOf(t.parent)+"/"+t.name:t.name;
 static Sprite SpriteAt(string path,string name)=>AssetDatabase.LoadAllAssetsAtPath(path).OfType<Sprite>().Single(x=>x.name==name);
 static SpriteRenderer Put(Transform root,Sprite sprite,float x,float y,int order,string name){var o=new GameObject(name);o.transform.SetParent(root,false);o.transform.position=new Vector3(x,y,0);var r=o.AddComponent<SpriteRenderer>();r.sprite=sprite;r.sortingOrder=order;return r;}
 static string SoilPrefix(Func<int,int,bool> inside){string join="";if(inside(0,1))join+="N";if(inside(1,0))join+="E";if(inside(0,-1))join+="S";if(inside(-1,0))join+="W";var absent=new System.Collections.Generic.List<string>();if(!(inside(-1,0)&&inside(0,1)&&inside(-1,1)))absent.Add("NW");if(!(inside(1,0)&&inside(0,1)&&inside(1,1)))absent.Add("NE");if(!(inside(1,0)&&inside(0,-1)&&inside(1,-1)))absent.Add("SE");if(!(inside(-1,0)&&inside(0,-1)&&inside(-1,-1)))absent.Add("SW");return "Farmland_DryTanBorder_SoilJoin_"+(join==""?"None":join)+"_AbsentCorners_"+(absent.Count==0?"None":string.Join("_",absent))+"_Variant";}
 static void Capture(Camera cam,string file,int w=1600,int h=900){var rt=new RenderTexture(w,h,24,RenderTextureFormat.ARGB32);rt.Create();var previous=RenderTexture.active;try{cam.aspect=(float)w/h;cam.targetTexture=rt;SortingGroup.UpdateAllSortingGroups();RenderPipeline.SubmitRenderRequest(cam,new UniversalRenderPipeline.SingleCameraRequest{destination=rt});RenderTexture.active=rt;var t=new Texture2D(w,h,TextureFormat.RGB24,false);t.ReadPixels(new Rect(0,0,w,h),0,0);t.Apply();File.WriteAllBytes(Out+file+".png",t.EncodeToPNG());Object.DestroyImmediate(t);}finally{cam.targetTexture=null;RenderTexture.active=previous;rt.Release();Object.DestroyImmediate(rt);}}
 static System.Collections.Generic.List<Vector3Int> Cells(Tilemap map){var result=new System.Collections.Generic.List<Vector3Int>();foreach(var c in map.cellBounds.allPositionsWithin)if(map.HasTile(c))result.Add(c);return result;}

 static JObject Describe(Renderer r){var sr=r as SpriteRenderer;var tm=r as TilemapRenderer;var mats=new JArray(r.sharedMaterials.Where(m=>m).Select(m=>new JObject{["name"]=m.name,["asset"]=AssetDatabase.GetAssetPath(m),["shader"]=m.shader.name,["queue"]=m.renderQueue,["renderType"]=m.GetTag("RenderType",false)}));return new JObject{["path"]=PathOf(r.transform),["type"]=r.GetType().Name,["sprite"]=sr&&sr.sprite?sr.sprite.name:"",["position"]=V(r.transform.position),["order"]=r.sortingOrder,["layer"]=r.sortingLayerName,["layerId"]=r.sortingLayerID,["sortPoint"]=sr?sr.spriteSortPoint.ToString():"",["tileMode"]=tm?tm.mode.ToString():"",["materials"]=mats};}

 static Tilemap Decor(Scene s)=>s.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<Tilemap>(true)).Single(x=>x.name=="BG_Decor 0");
 static Camera CameraIn(Scene s)=>s.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<Camera>(true)).First();
 static SpriteRenderer RestoredTree(Transform parent,Tilemap map,Vector3Int cell){var sprite=SpriteAt("Assets/Art/Packs/Cute_Fantasy/Crops/Fruit_Tree_Stages.png","FruitTree_Growth_YoungTree");var anchor=map.CellToWorld(cell)+map.CellToLocalInterpolated(map.tileAnchor)-map.CellToLocal(Vector3Int.zero);var sr=Put(parent,sprite,anchor.x,anchor.y,3,"Restored scenery tree - original cell "+cell);sr.spriteSortPoint=SpriteSortPoint.Pivot;sr.sharedMaterial=map.GetComponent<TilemapRenderer>().sharedMaterial;return sr;}
 static SpriteRenderer[] Restore(Scene s){var map=Decor(s);var root=new GameObject("RESTORED scenery trees - ground depth - preview only");SceneManager.MoveGameObjectToScene(root,s);return new[]{RestoredTree(root.transform,map,new Vector3Int(13,-13,0)),RestoredTree(root.transform,map,new Vector3Int(23,-5,0))};}
 static void GroundSortOrchard(Scene s){foreach(var sr in s.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<SpriteRenderer>(true)).Where(r=>r.name.StartsWith("Orchard tree"))){var sprite=sr.sprite;var anchor=sr.transform.position-new Vector3((sprite.pivot.x-16)/16,(sprite.pivot.y-16)/16,0);var g=new GameObject("Tree ground anchor - "+sr.name);g.transform.SetParent(sr.transform.parent,false);g.transform.position=anchor;var group=g.AddComponent<SortingGroup>();group.sortingLayerName="Default";group.sortingOrder=3;sr.transform.SetParent(g.transform,true);sr.sortingLayerName="Default";sr.sortingOrder=0;sr.spriteSortPoint=SpriteSortPoint.Pivot;}}
 static void Fixture(){WithScenes(()=>{var scene=Open("Assets/Editor/FarmingPreviewsV4/A3-southeast-clearing.unity");try{var map=Decor(scene);var tile=AssetDatabase.LoadAssetAtPath<Tile>("Assets/Tilemaps/Tiles/FruitTree_Growth_YoungTree.asset");var cam=CameraIn(scene);cam.transform.position=new Vector3(-61.5f,-11.5f,-10);cam.orthographicSize=4;map.SetTile(new Vector3Int(13,-13,0),tile);map.SetTile(new Vector3Int(23,-5,0),tile);Capture(cam,"south-before-depth");cam.transform.position=new Vector3(-52,-4.5f,-10);Capture(cam,"east-before-depth");map.SetTile(new Vector3Int(13,-13,0),null);map.SetTile(new Vector3Int(23,-5,0),null);var restored=Restore(scene);cam.transform.position=new Vector3(-61.5f,-11.5f,-10);Capture(cam,"south-after-depth");cam.transform.position=new Vector3(-52,-4.5f,-10);Capture(cam,"east-after-depth");cam.transform.position=new Vector3(-61.5f,-11.5f,-10);restored[0].transform.position=new Vector3(-61.5f,-10.9375f,0);Capture(cam,"north-ground-fixture");restored[0].enabled=false;Capture(cam,"fence-only-fixture");restored[0].enabled=true;foreach(var sr in scene.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<SpriteRenderer>(true)).Where(r=>r.sprite&&(r.sprite.name.StartsWith("Fence_")||r.sprite.name.StartsWith("Gate_"))))sr.enabled=false;Capture(cam,"north-tree-only-fixture");restored[0].transform.position=new Vector3(-61.5f,-12.5f,0);Capture(cam,"south-tree-only-fixture");File.WriteAllText(Out+"fixture.json",new JObject{["southFoot"]=new JArray(-61.5,-12.5),["northFoot"]=new JArray(-61.5,-10.9375),["fenceRowY"]=-11,["southTree"]=Describe(restored[0]),["tilemap"]=Describe(map.GetComponent<TilemapRenderer>()),["notSaved"]=true}.ToString());}finally{Close(scene);}});}
 static Scene Open(string path){var s=EditorSceneManager.OpenScene(path,OpenSceneMode.Additive);SceneManager.SetActiveScene(s);EditorSceneManager.SetSceneCullingMask(s,1UL<<41);CameraIn(s).overrideSceneCullingMask=1UL<<41;return s;}
 static void Close(Scene s){EditorSceneManager.CloseScene(s,true);}
 static void WithScenes(Action body){var main=SceneManager.GetActiveScene();Assert(main.path=="Assets/Scenes/Main.unity"&&!main.isDirty,"Main must be clean");var mask=EditorSceneManager.GetSceneCullingMask(main);var guards=main.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<Light2D>(true)).Select(x=>new{light=x,state=x.enabled}).ToArray();try{foreach(var g in guards)g.light.enabled=false;EditorSceneManager.SetSceneCullingMask(main,1UL<<40);body();}finally{SceneManager.SetActiveScene(main);EditorSceneManager.SetSceneCullingMask(main,mask);foreach(var g in guards)g.light.enabled=g.state;File.WriteAllText(Out+"main-dirty.txt",main.isDirty.ToString());}}
 static void Reposition(Scene s,int count,bool mixed){float[] xs=count==6?new[]{-60f,-56f,-52f}:new[]{-60f,-57f,-54f,-51f};var root=s.GetRootGameObjects().Single(o=>o.name.StartsWith("OPTIONAL Orchard Capacity"));foreach(var sr in root.GetComponentsInChildren<SpriteRenderer>()){var m=System.Text.RegularExpressions.Regex.Match(sr.name,@"row(\d+) col(\d+)");Assert(m.Success,"Unexpected orchard sprite name: "+sr.name);int row=int.Parse(m.Groups[1].Value),col=int.Parse(m.Groups[2].Value);float x=xs[col],y=-13.5f-row*3.5f;if(sr.name.StartsWith("Orchard soil")){int ix=sr.name[sr.name.Length-2]-'0',iy=sr.name[sr.name.Length-1]-'0';sr.transform.position=new Vector3(x-.5f+ix,y-.5f+iy,0);}else{if(mixed){string[] names={"FruitTree_Growth_YoungTree","FruitTree_Growth_SmallSapling","FruitTree_Growth_MatureTree","Apple_Tree_Fruiting_WholeTree"};var name=names[(row*xs.Length+col)%4];sr.sprite=SpriteAt("Assets/Art/Packs/Cute_Fantasy/Crops/"+(name.StartsWith("Apple")?"Apple_Tree.png":"Fruit_Tree_Stages.png"),name);}sr.transform.position=new Vector3(x+(sr.sprite.pivot.x-16)/16,y+(sr.sprite.pivot.y-16)/16,0);}}}
 static void CaptureViews(Scene s,string name){var cam=CameraIn(s);foreach(var v in new[]{new{name="town",x=-70f,y=-10f,size=32f},new{name="normal",x=-61f,y=-11f,size=18f},new{name="north-normal",x=-67f,y=0f,size=18f},new{name="south-normal",x=-61f,y=-23f,size=18f},new{name="south-close",x=-58f,y=-27f,size=9f},new{name="north-close",x=-60f,y=-.5f,size=12f},new{name="orchard-close",x=-57.5f,y=-16f,size=9f},new{name="fence-detail",x=-57.5f,y=-11.5f,size=4f}}){cam.transform.position=new Vector3(v.x,v.y,-10);cam.orthographicSize=v.size;Capture(cam,"renders/"+name+"-"+v.name);}}
 static void OrchardControl(){WithScenes(()=>{var scene=Open("Assets/Editor/FarmingOrchardCapacityPreview/G6-one-tile-up.unity");try{var cam=CameraIn(scene);cam.transform.position=new Vector3(-57.5f,-11.5f,-10);cam.orthographicSize=4;Capture(cam,"orchard-depth-with-fence");foreach(var sr in scene.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<SpriteRenderer>(true)).Where(r=>r.sprite&&(r.sprite.name.StartsWith("Fence_")||r.sprite.name.StartsWith("Gate_"))))sr.enabled=false;Capture(cam,"orchard-depth-without-fence");}finally{Close(scene);}});}
 static JObject Art(Sprite s,Matrix4x4 matrix,string path,int order,string layer,string kind){AssetDatabase.TryGetGUIDAndLocalFileIdentifier(s,out string guid,out long id);var m=new JArray();for(int r=0;r<4;r++)for(int c=0;c<4;c++)m.Add(matrix[r,c]);return new JObject{["path"]=path,["sprite"]=s.name,["asset"]=AssetDatabase.GetAssetPath(s),["guid"]=guid,["fileId"]=id,["rect"]=new JArray(s.rect.x,s.rect.y,s.rect.width,s.rect.height),["pivot"]=new JArray(s.pivot.x,s.pivot.y),["ppu"]=s.pixelsPerUnit,["matrix"]=m,["order"]=order,["layer"]=layer,["kind"]=kind};}
 static JArray AuditArt(GameObject[] roots){var art=new JArray();foreach(var r in roots.SelectMany(o=>o.GetComponentsInChildren<SpriteRenderer>(true)).Where(x=>x.gameObject.activeInHierarchy&&x.sprite)){if(r.bounds.max.x< -78||r.bounds.min.x> -44||r.bounds.max.y< -39||r.bounds.min.y>12)continue;var a=Art(r.sprite,r.transform.localToWorldMatrix,PathOf(r.transform),r.sortingOrder,r.sortingLayerName,"SpriteRenderer");a["flipX"]=r.flipX;a["flipY"]=r.flipY;art.Add(a);}foreach(var map in roots.SelectMany(o=>o.GetComponentsInChildren<Tilemap>(true)).Where(x=>x.gameObject.activeInHierarchy)){var render=map.GetComponent<TilemapRenderer>();foreach(var cell in Cells(map)){var s=map.GetSprite(cell);if(!s)continue;var matrix=map.transform.localToWorldMatrix*Matrix4x4.Translate(map.CellToLocalInterpolated((Vector3)cell+map.tileAnchor))*map.GetTransformMatrix(cell);var v=matrix.MultiplyPoint3x4(Vector3.zero);if(v.x< -78||v.x> -44||v.y< -39||v.y>12)continue;var a=Art(s,matrix,PathOf(map.transform),render.sortingOrder,render.sortingLayerName,"Tilemap");a["cell"]=V(cell);a["worldAnchor"]=V(v);a["tile"]=AssetDatabase.GetAssetPath(map.GetTile(cell));art.Add(a);}}return art;}

 static void Recapture(){var records=new JArray();WithScenes(()=>{foreach(var folder in new[]{"Assets/Editor/FarmingPreviewsV4","Assets/Editor/FarmingOrchardRowsPreview","Assets/Editor/FarmingOrchardCapacityPreview"})foreach(var path in Directory.GetFiles(folder,"*.unity")){var scene=Open(path);try{CaptureViews(scene,Path.GetFileNameWithoutExtension(path));var roots=scene.GetRootGameObjects();records.Add(new JObject{["scene"]=path,["art"]=AuditArt(roots),["colliders"]=new JArray(roots.SelectMany(o=>o.GetComponentsInChildren<Collider2D>(true)).Select(c=>new JObject{["path"]=PathOf(c.transform),["type"]=c.GetType().Name,["enabled"]=c.enabled,["trigger"]=c.isTrigger,["min"]=V(c.bounds.min),["max"]=V(c.bounds.max)}))});}finally{Close(scene);}}});File.WriteAllText(Out+"final-scene-art.json",records.ToString());}
 static void BuildAll(){Directory.CreateDirectory(Out+"renders");var records=new JArray();var logs=new JArray();Application.LogCallback logger=(m,st,t)=>{if(t==LogType.Warning||t==LogType.Error||t==LogType.Exception)logs.Add(new JObject{["type"]=t.ToString(),["message"]=m});};Application.logMessageReceived+=logger;try{WithScenes(()=>{string[] names={"A0-existing","A1-first-beds","A2-stump-reclamation","A2b-southeast-work","A3-southeast-clearing","B3-terraced-rows","C3-grove-courtyard"};foreach(var n in names){string path="Assets/Editor/FarmingPreviewsV4/"+n+".unity";var s=Open(path);try{if(!n.StartsWith("A0")&&!n.StartsWith("A1")){Restore(s);EditorSceneManager.SaveScene(s,path);}CaptureViews(s,n);records.Add(new JObject{["scene"]=path,["restoredTrees"]=n.StartsWith("A0")||n.StartsWith("A1")?0:2,["mainComponentsNotEdited"]=true});}finally{Close(s);}}
 foreach(var path in Directory.GetFiles("Assets/Editor/FarmingOrchardRowsPreview","*.unity")){var s=Open(path);try{Restore(s);GroundSortOrchard(s);EditorSceneManager.SaveScene(s,path);CaptureViews(s,Path.GetFileNameWithoutExtension(path));records.Add(new JObject{["scene"]=path,["restoredTrees"]=2});}finally{Close(s);}}
 foreach(var path in Directory.GetFiles("Assets/Editor/FarmingOrchardCapacityPreview","*.unity").Where(p=>!p.Contains("-mixed-up"))){var s=Open(path);try{Restore(s);if(path.Contains("one-tile-up"))Reposition(s,path.Contains("G6")?6:8,false);GroundSortOrchard(s);EditorSceneManager.SaveScene(s,path);CaptureViews(s,Path.GetFileNameWithoutExtension(path));records.Add(new JObject{["scene"]=path,["restoredTrees"]=2,["originalCrops"]=s.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<SpriteRenderer>(true)).Count(r=>r.name.StartsWith("Crop ")),["paths"]=Cells(s.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<Tilemap>(true)).Single(x=>x.name=="-2 Old Path")).Count,["trees"]=new JArray(s.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<SpriteRenderer>(true)).Where(r=>r.name.StartsWith("Orchard tree")||r.name.StartsWith("Restored scenery tree")).Select(r=>new JObject{["renderer"]=Describe(r),["boundsMin"]=V(r.bounds.min),["boundsMax"]=V(r.bounds.max),["ground"]=V(r.GetComponentInParent<SortingGroup>()?r.GetComponentInParent<SortingGroup>().transform.position:r.transform.position)}))});}finally{Close(s);}}
 foreach(int count in new[]{6,8}){string prefix=count==6?"G6":"H8",path="Assets/Editor/FarmingOrchardCapacityPreview/"+prefix+"-one-tile-up.unity";var s=Open(path);try{Reposition(s,count,true);var dest="Assets/Editor/FarmingOrchardCapacityPreview/"+prefix+"-mixed-up.unity";EditorSceneManager.SaveScene(s,dest);CaptureViews(s,prefix+"-mixed-up");records.Add(new JObject{["scene"]=dest,["treeCount"]=count,["fullTileUp"]=true});}finally{Close(s);}}
 });File.WriteAllText(Out+"build-records.json",records.ToString());}finally{Application.logMessageReceived-=logger;File.WriteAllText(Out+"render-logs.json",logs.ToString());}}
}
