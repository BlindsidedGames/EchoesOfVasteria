    static string JoinPrefix(string material,Func<int,int,bool> inside) {
        string join="";if(inside(0,1))join+="N";if(inside(1,0))join+="E";if(inside(0,-1))join+="S";if(inside(-1,0))join+="W";
        var absent=new System.Collections.Generic.List<string>();
        if(!(inside(-1,0)&&inside(0,1)&&inside(-1,1)))absent.Add("NW");
        if(!(inside(1,0)&&inside(0,1)&&inside(1,1)))absent.Add("NE");
        if(!(inside(1,0)&&inside(0,-1)&&inside(1,-1)))absent.Add("SE");
        if(!(inside(-1,0)&&inside(0,-1)&&inside(-1,-1)))absent.Add("SW");
        return material+"_SoilJoin_"+(join==""?"None":join)+"_AbsentCorners_"+(absent.Count==0?"None":string.Join("_",absent))+"_Variant";
    }
    static void RefineGroundAndPaths(GameObject town,int stage) {
        // The currently visible thin brown town track is WetOverlay, not bright-green dirt fill.
        var tm=town.GetComponentsInChildren<Tilemap>(true).First(x=>x.name=="-2 Old Path");
        var tileAssets=AssetDatabase.FindAssets("t:Tile",new[]{"Assets/Tilemaps/Tiles"}).Select(g=>AssetDatabase.LoadAssetAtPath<Tile>(AssetDatabase.GUIDToAssetPath(g))).Where(t=>t!=null && t.name.StartsWith("Farmland_WetOverlay_")).ToArray();
        var changed=new System.Collections.Generic.HashSet<Vector3Int>();
        Action<int,int> add=(x,y)=>changed.Add(tm.WorldToCell(new Vector3(x,y,0)));
        // Original two-column gate approach is already authored: world columns
        // -62/-61 at Y=-3/-2, centered on gate(-61,-1). Preserve it exactly.
        // Later gates receive the same two adjacent columns; do not create two gates.
        if(stage>=2){for(int x=-60;x<=-58;x++)add(x,-3);for(int x=-59;x<=-58;x++)for(int y=-4;y<=-3;y++)add(x,y);}
        if(stage>=3){int right=town.name.Contains("A3")?-58:-57;
            for(int x=-60;x<=right;x++)add(x,-22);
            for(int x=right-1;x<=right;x++)for(int y=-23;y<=-22;y++)add(x,y);
        }
        foreach(var cell in changed)tm.SetTile(cell,tileAssets.First());
        var affected=new System.Collections.Generic.HashSet<Vector3Int>(changed);
        foreach(var cell in changed)for(int dx=-1;dx<=1;dx++)for(int dy=-1;dy<=1;dy++)affected.Add(cell+new Vector3Int(dx,dy,0));
        foreach(var cell in affected) {
            if(tm.GetTile(cell)==null)continue;
            Func<int,int,bool> occupied=(dx,dy)=>tm.GetTile(cell+new Vector3Int(dx,dy,0))!=null;
            var prefix=JoinPrefix("Farmland_WetOverlay",occupied);
            var tile=tileAssets.First(t=>t.name.StartsWith(prefix));tm.SetTile(cell,tile);tm.SetTileFlags(cell,TileFlags.None);tm.SetTransformMatrix(cell,Matrix4x4.identity);
        }
        // Remove small ground clutter only inside the proposed enclosures, preserving surrounding woodland.
        var decor=town.GetComponentsInChildren<Tilemap>(true).First(x=>x.name=="BG_Decor 0");
        foreach(var cell in decor.cellBounds.allPositionsWithin) {
            var p=decor.GetCellCenterWorld(cell);bool inField=p.x>-64 && p.x<-54 && p.y>-1 && p.y<6;
            inField|=stage>=2 && p.x>-64 && p.x<-52 && p.y>-11 && p.y<-4;
            inField|=stage>=3 && p.x>-65 && p.x<-50 && p.y>-33 && p.y<-23;
            if(inField)decor.SetTile(cell,null);
        }
        Func<Vector3,bool> within=p=> (p.x>-64 && p.x<-54 && p.y>-1 && p.y<6)
            || (stage>=2 && p.x>-64 && p.x<-52 && p.y>-11 && p.y<-4)
            || (stage>=3 && p.x>(town.name.Contains("C3")?-65:-63) && p.x<(town.name.Contains("A3")?-53:town.name.Contains("B3")?-51:-50) && p.y>(town.name.Contains("A3")?-29:-33) && p.y<-23);
        var clutter=new JArray();
        foreach(var sr in town.GetComponentsInChildren<SpriteRenderer>(true)) {
            if(sr.sprite==null || !sr.gameObject.activeInHierarchy)continue;
            bool prop=sr.sprite.name.StartsWith("Outdoor_Decor_") || sr.sprite.name.StartsWith("Crate_");
            var b=sr.bounds;var plots=new System.Collections.Generic.List<Bounds>{new Bounds(new Vector3(-59,2.5f,0),new Vector3(10.9f,7.9f,10))};
            if(stage>=2)plots.Add(new Bounds(new Vector3(-58,-7.5f,0),new Vector3(12.9f,7.9f,10)));
            if(stage>=3)plots.Add(new Bounds(town.name.Contains("A3")?new Vector3(-58,-26,0):town.name.Contains("B3")?new Vector3(-57,-28,0):new Vector3(-57.5f,-28,0),town.name.Contains("A3")?new Vector3(10.9f,6.9f,10):town.name.Contains("B3")?new Vector3(12.9f,10.9f,10):new Vector3(15.9f,10.9f,10)));
            bool overlap=plots.Any(plot=>plot.Intersects(b));
            if(prop && overlap) {clutter.Add(new JObject{["path"]=PathOf(sr.transform),["sprite"]=sr.sprite.name,["position"]=V(sr.transform.position)});sr.gameObject.SetActive(false);}
        }
        var gates=new JArray();
        Action<int,int,bool,int,int> prove=(gx,gy,north,ax,ay)=> {
            var starts=new[]{tm.WorldToCell(new Vector3(ax,ay,0)),tm.WorldToCell(new Vector3(ax,ay,0))};
            var approaches=new JArray();
            for(int side=0;side<2;side++) {
                int x=gx-1+side;int exteriorY=north?gy:gy-1;
                var target=tm.WorldToCell(new Vector3(x,exteriorY,0));
                var seen=new System.Collections.Generic.HashSet<Vector3Int>();var pending=new System.Collections.Generic.Queue<Vector3Int>();pending.Enqueue(starts[side]);
                while(pending.Count>0){var c=pending.Dequeue();if(!seen.Add(c)||tm.GetTile(c)==null)continue;foreach(var d in new[]{Vector3Int.up,Vector3Int.down,Vector3Int.left,Vector3Int.right})if(!seen.Contains(c+d)&&tm.GetTile(c+d)!=null)pending.Enqueue(c+d);}
                bool open=!town.scene.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<SpriteRenderer>(true)).Any(r=>r.gameObject.activeInHierarchy&&r.name==(north?"North fence":"South fence")&&Mathf.Abs(r.transform.position.y-gy)<0.01f&&Mathf.Abs(r.transform.position.x-(gx+(side==0?-0.5f:0.5f)))<0.6f);
                if(!seen.Contains(target)||!open)throw new Exception("Dual-column gate route failed at "+gx+","+gy+" lane "+side);
                approaches.Add(new JObject{["column"]=x,["exteriorCellCenter"]=V(tm.GetCellCenterWorld(target)),["connectedToExistingTrack"]=true,["touchesGateBoundary"]=true,["fenceApertureOpen"]=open});
            }
            gates.Add(new JObject{["center"]=new JArray(gx,gy),["dualColumns"]=approaches,["connectedToExistingTrack"]=true,["fenceApertureOpen"]=true,["runtimeNavigationTested"]=false});
        };
        prove(-61,-1,false,-63,-3);
        if(stage>=2)prove(-58,-4,true,-64,-3);
        if(stage>=3)prove(town.name.Contains("A3")?-58:-57,-23,true,-60,-22);
        var info=new JObject{["clearedGroundProps"]=clutter,["pathTilemap"]=PathOf(tm.transform),["active"]=tm.gameObject.activeInHierarchy,["material"]="Farmland_WetOverlay",["cellsAdded"]=new JArray(changed.Select(c=>V(tm.GetCellCenterWorld(c)))),["gates"]=gates};
        File.WriteAllText(Out+town.name.Replace("Preview Town - ","")+"-path-proof.json",info.ToString());
    }
    static void Capture(Camera cam,string path,int width,int height)
    {

        try {
            cam.aspect=(float)width/height;cam.targetTexture=rt;
            RenderPipeline.SubmitRenderRequest(cam,new UniversalRenderPipeline.SingleCameraRequest{destination=rt});
            RenderTexture.active=rt;var tex=new Texture2D(width,height,TextureFormat.RGB24,false);tex.ReadPixels(new Rect(0,0,width,height),0,0);tex.Apply();File.WriteAllBytes(path,tex.EncodeToPNG());UnityEngine.Object.DestroyImmediate(tex);
        } finally {cam.targetTexture=null;RenderTexture.active=previous;rt.Release();UnityEngine.Object.DestroyImmediate(rt);}
    }

    static void AuditReferences() {
        var main=SceneManager.GetActiveScene();if(EditorApplication.isPlayingOrWillChangePlaymode || main.path!="Assets/Scenes/Main.unity" || main.isDirty)throw new Exception("Main must be clean");
        var guards=main.GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<Light2D>(true)).Select(x=>new {light=x,enabled=x.enabled,path=PathOf(x.transform)}).ToArray();var preview=default(Scene);
        Func<Scene,Tilemap> pathMap=sc=>sc.GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<Tilemap>(true)).First(x=>x.name=="-2 Old Path");
        Func<Tilemap,JArray> approach=tm=>new JArray(new[]{new Vector3Int(-62,-3,0),new Vector3Int(-61,-3,0),new Vector3Int(-62,-2,0),new Vector3Int(-61,-2,0)}.Select(w=>{var c=tm.WorldToCell(w);var tile=tm.GetTile(c);return new JObject{["world"]=V(w),["cell"]=V(c),["asset"]=AssetDatabase.GetAssetPath(tile),["name"]=tile.name,["transform"]=tm.GetTransformMatrix(c).ToString()};}));
        Func<Scene,JArray> poses=sc=>new JArray(sc.GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<SpriteRenderer>(true)).Where(x=>x.gameObject.activeInHierarchy&&x.sprite!=null&&(x.sprite.name.StartsWith("Windmill")||x.sprite.name.StartsWith("Flora_")||x.sprite.name.StartsWith("Tillman_"))).Select(x=>new JObject{["sprite"]=x.sprite.name,["position"]=V(x.transform.position),["scale"]=V(x.transform.lossyScale),["order"]=x.sortingOrder,["asset"]=AssetDatabase.GetAssetPath(x.sprite)}));
        try {foreach(var g in guards)g.light.enabled=false;preview=EditorSceneManager.OpenScene("Assets/Editor/FarmingPreviewsV4/A3-southeast-clearing.unity",OpenSceneMode.Additive);
            var before=approach(pathMap(main));var after=approach(pathMap(preview));var mainPoses=poses(main);var newPoses=poses(preview);
            var identical=JToken.DeepEquals(before,after);var sameGeometry=before.Zip(after,(a,b)=>JToken.DeepEquals(a["world"],b["world"])&&JToken.DeepEquals(a["cell"],b["cell"])&&JToken.DeepEquals(a["transform"],b["transform"])).All(x=>x);
            if(!sameGeometry||!JToken.DeepEquals(mainPoses,newPoses))throw new Exception("Original approach geometry/pose mismatch");
            var root=preview.GetRootGameObjects().First(x=>x.name.StartsWith("CONCEPT Farm Layout"));var railTop=root.GetComponentsInChildren<SpriteRenderer>().Where(x=>x.name=="North fence"&&Mathf.Abs(x.transform.position.y-6)<0.01f).Max(x=>x.bounds.max.y);
            File.WriteAllText(Out+"layout-reference.json",new JObject{["originalApproachGeometryUnchanged"]=sameGeometry,["originalApproachTilesIdentical"]=identical,["originalApproach"]=before,["previewApproach"]=after,["originalPosesUnchanged"]=true,["originalPoses"]=mainPoses,["previewPoses"]=newPoses,["originalFarmSouthGate"]=new JArray(-61,-1),["originalBackFenceY"]=10,["revisedBackFenceY"]=6,["revisedFenceBoundsTop"]=railTop,["noOriginalFarmPathExtension"]=true,["mainDirty"]=main.isDirty,["playing"]=EditorApplication.isPlaying}.ToString());
        }finally {if(preview.IsValid())EditorSceneManager.CloseScene(preview,true);SceneManager.SetActiveScene(main);foreach(var g in guards)g.light.enabled=g.enabled;File.WriteAllText(Out+"reference-light-isolation.json",new JObject{["mainDirtyAfterRestore"]=main.isDirty,["guards"]=new JArray(guards.Select(x=>new JObject{["path"]=x.path,["before"]=x.enabled,["after"]=x.light.enabled}))}.ToString());}
    }
