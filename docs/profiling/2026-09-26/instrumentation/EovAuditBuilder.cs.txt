using System;using System.IO;using System.Linq;using UnityEditor;using UnityEditor.Build;using UnityEditor.Build.Reporting;
public static class EovAuditBuilder {
 public static void Build(){PlayerSettings.productName="EOV Performance Audit Disposable";PlayerSettings.SetScriptingBackend(NamedBuildTarget.Standalone,ScriptingImplementation.Mono2x);PlayerSettings.runInBackground=true;PlayerSettings.enableFrameTimingStats=true;var args=Environment.GetCommandLineArgs();int i=Array.IndexOf(args,"-buildOutput");string dest=i>=0?args[i+1]:Path.GetFullPath("Build/EovAudit.exe");var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=EditorBuildSettings.scenes.Where(s=>s.enabled).Select(s=>s.path).ToArray(),locationPathName=dest,target=BuildTarget.StandaloneWindows64,options=BuildOptions.Development|BuildOptions.DetailedBuildReport});File.WriteAllText(Path.Combine(Path.GetDirectoryName(dest),"build-summary.txt"),$"{report.summary.result}\n{report.summary.totalTime}\n{report.summary.totalSize}\n{report.summary.totalErrors} errors\nMono Development; separate product and save root; Steam and UGS disabled in disposable copy");if(report.summary.result!=BuildResult.Succeeded)throw new Exception("Audit player build failed");}
}
