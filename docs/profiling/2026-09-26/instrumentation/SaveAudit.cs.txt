using System;using System.IO;using System.Linq;using System.Collections.Generic;using System.Threading.Tasks;using System.Diagnostics;using UnityEngine;using Blindsided;using Blindsided.SaveData;
public static class SaveAudit2 {
 static object s;static Task pending;static List<string> lines=new List<string>{"profile,index,capture_main_ms,save_complete_ms,load_complete_ms,status,load_status"};static int index;static string stage;static double next;static GameData detached;static object Inv(string n,params object[] a)=>s.GetType().GetMethod(n).Invoke(s,a);
 static async Task Measure(){var d=Oracle.oracle.saveData;var sw=Stopwatch.StartNew();Blindsided.EventHandler.SaveData(true);if(stage=="history10000"){detached=Sirenix.Serialization.SerializationUtility.DeserializeValue<GameData>(Sirenix.Serialization.SerializationUtility.SerializeValue(d,Sirenix.Serialization.DataFormat.Binary),Sirenix.Serialization.DataFormat.Binary);var template=detached.General.RecentRuns.FirstOrDefault()??new GameData.RunRecord();detached.General.RecentRuns=Enumerable.Range(0,10000).Select(_=>new GameData.RunRecord{MapType=template.MapType,Distance=template.Distance,Duration=template.Duration,TasksCompleted=template.TasksCompleted,EnemiesKilled=template.EnemiesKilled}).ToList();d=detached;sw.Restart();}var task=SaveManager.Instance.SaveDetailedAsync(d,(stage=="fresh"?"Save1":stage=="developed"?"Save2":"Save3"));double main=sw.Elapsed.TotalMilliseconds;var result=await task;double total=sw.Elapsed.TotalMilliseconds;sw.Restart();var loaded=await SaveManager.Instance.LoadDetailedAsync((stage=="fresh"?"Save1":stage=="developed"?"Save2":"Save3"));double read=sw.Elapsed.TotalMilliseconds;lines.Add($"{stage},{index++},{main:R},{total:R},{read:R},{result.Status},{loaded.Status}");}
 public static object Main(){s=AppDomain.CurrentDomain.GetData("EOV_AUDIT");foreach(var profile in new[]{"fresh","developed","history10000"}){string p=profile;Inv("Add","save-load-"+p,(Action)(()=>{Inv("Reset",p=="fresh"?"fresh":"developed",false);stage=p;next=0;index=0;}),25.0,(Action)(()=>{if(Time.realtimeSinceStartupAsDouble<next||pending!=null&&!pending.IsCompleted)return;next=Time.realtimeSinceStartupAsDouble+1;pending=Measure();}),5.0);}
 Inv("Add","save-results",(Action)(()=>File.WriteAllLines("docs/profiling/2026-09-26/save-latency.csv",lines)),5.0,null,1.0);
 foreach(double away in new[]{3600d,86400d,604800d}){double a=away;Inv("Add","offline-"+away,(Action)(()=>{Inv("Reset","developed",false);var d=Oracle.oracle.saveData;double now=DateTime.UtcNow.Subtract(DateTime.UnixEpoch).TotalSeconds;foreach(var r in d.Disciples.Values)r.LastGenerationTime=now-a;d.DateQuitString=DateTime.UtcNow.AddSeconds(-a).ToString(System.Globalization.CultureInfo.InvariantCulture);var sw=Stopwatch.StartNew();Oracle.AwayForSeconds();sw.Stop();File.AppendAllText("docs/profiling/2026-09-26/offline-latency.csv",a+","+sw.Elapsed.TotalMilliseconds.ToString("R",System.Globalization.CultureInfo.InvariantCulture)+Environment.NewLine);}),15.0,null,0.0);}
 return "Disposable durable save/load and offline-processing scenarios appended";}
}
