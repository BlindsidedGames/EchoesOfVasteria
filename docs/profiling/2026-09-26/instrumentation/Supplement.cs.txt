using System;using System.IO;using System.Linq;using System.Reflection;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Blindsided;using Blindsided.SaveData;using Blindsided.Utilities;using TimelessEchoes;using TimelessEchoes.Gear;using TimelessEchoes.Gear.UI;using TimelessEchoes.Upgrades;
public static class SupplementAudit {
 const BindingFlags F=BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Instance|BindingFlags.Static;static object s;static Oracle o;static ForgeWindowUI forge;static string slot;static FeedbackForm feedback;static GameObject guard;static string[] keys;static bool[] exists;static float[] nums;static string date;static bool active;static int snapshots,operations;static double opMs;static readonly string dir="docs/profiling/2026-09-26/";
 static object Get(object obj,string n)=>obj.GetType().GetField(n,F).GetValue(obj);static void Set(object obj,string n,object v)=>obj.GetType().GetField(n,F).SetValue(obj,v);static object Call(object obj,string n,params object[] args)=>obj.GetType().GetMethods(F).First(m=>m.Name==n&&m.GetParameters().Length==args.Length).Invoke(obj,args);static object Inv(string n,params object[] args)=>Call(s,n,args);static void Add(string n,Action setup,double sec,Action tick=null,double warm=5)=>Inv("Add",n,setup,sec,tick,warm);
 static void Saved(){snapshots++;File.AppendAllText(dir+"autosave-events.csv",Time.realtimeSinceStartupAsDouble+","+snapshots+Environment.NewLine);}
 static void Cleanup(){if(!active)return;Call(o,"StopAutosaveLoop");Call(o,"CompleteActiveSaveBlocking");Call(o,"StopSaveCoordinatorMonitoring");Set(o,"_savePending",false);Set(o,"_recoveryRequired",true);Blindsided.EventHandler.OnSaveData-=Saved;for(int i=0;i<3;i++){if(!exists[i])PlayerPrefs.DeleteKey(keys[i]);else if(i==2)PlayerPrefs.SetString(keys[i],date);else PlayerPrefs.SetFloat(keys[i],nums[i]);}SaveManager.Instance.SetCurrentSlot(slot);FeedbackForm.Instance=feedback;if(guard!=null)UnityEngine.Object.DestroyImmediate(guard);active=false;File.WriteAllText(dir+"autosave-cleanup.txt","Completed pending disposable save; recovery guard restored; slot metadata PlayerPrefs and SaveManager slot restored; feedback network guard removed. Snapshots="+snapshots);}
 public static object Main(){s=AppDomain.CurrentDomain.GetData("EOV_AUDIT");o=Oracle.oracle;forge=UnityEngine.Object.FindObjectsByType<ForgeWindowUI>(FindObjectsInactive.Include).First();if(!(bool)Get(o,"_recoveryRequired"))throw new Exception("Save guard absent");var root=(string)typeof(SaveManager).GetField("rootPathOverride",F).GetValue(null);if(root!=Path.GetFullPath("Library/PerformanceAudit/disposable"))throw new Exception("Wrong save root");int old=((System.Collections.IList)Get(s,"queue")).Count;Set(s,"qi",old-1);
 foreach(var field in new[]{"forge","inventory",(string)null}){var panel=field;string name="equipment-events-"+(panel??"closed");Add(name,()=>{Inv("Reset","developed",false);foreach(var r in Resources.LoadAll<Resource>("Resource Items"))o.saveData.Resources[r.name]=new GameData.ResourceEntry{Amount=1e8,Earned=true,Tier=1};Blindsided.EventHandler.LoadData();Call(forge,"StopAutoCrafting");Inv("Panel",panel);operations=0;opMs=0;},15,()=>{var watch=System.Diagnostics.Stopwatch.StartNew();string sl=new[]{"Weapon","Chest","Helmet","Boots"}[operations%4];var item=CraftingService.Instance.Craft(Resources.Load<CoreSO>("Gear/Cores/Vastium"),sl,null,Resources.LoadAll<Resource>("Resource Items").First(r=>r.name=="Vastium Core"),1);if(item==null)throw new Exception("Craft failed");var previous=EquipmentController.Instance.GetEquipped(sl);if(!EquipmentController.Instance.Equip(item))throw new Exception("Equip failed");if(previous!=null)SalvageService.Instance.Salvage(previous,true);watch.Stop();operations++;opMs+=watch.Elapsed.TotalMilliseconds;File.AppendAllText(dir+"equipment-events.csv",name+","+operations+","+watch.Elapsed.TotalMilliseconds+Environment.NewLine);});}
 Add("autosave-coordinator-private-root",()=>{Inv("Reset","developed",false);CauldronManager.Instance.StopTasting();Call(forge,"StopAutoCrafting");slot=SaveManager.Instance.CurrentSlotName;keys=new[]{"Completion","Playtime","Date"}.Select(k=>o.GetSlotPlayerPrefsKey(o.CurrentSlot,k)).ToArray();exists=keys.Select(PlayerPrefs.HasKey).ToArray();nums=keys.Select(k=>PlayerPrefs.GetFloat(k)).ToArray();date=PlayerPrefs.GetString(keys[2]);feedback=FeedbackForm.Instance;guard=new GameObject("Disposable Feedback Network Guard");guard.hideFlags=HideFlags.HideAndDontSave;guard.SetActive(false);FeedbackForm.Instance=guard.AddComponent<FeedbackForm>();AppDomain.CurrentDomain.SetData("EOV_AUTOSAVE_WAS_RUNNING",Get(o,"_autosaveRoutine")!=null);active=true;AppDomain.CurrentDomain.SetData("EOV_SUPPLEMENT_CLEANUP",(Action)Cleanup);snapshots=0;Blindsided.EventHandler.OnSaveData+=Saved;Set(o,"_recoveryRequired",false);Call(o,"StartAutosaveLoop",10f);},75);
 Add("autosave-coalesced-burst",()=>{for(int i=0;i<10;i++)Call(o,"RequestSave");},5,null,0);
 Add("supplement-cleanup",()=>{Cleanup();Inv("Reset","developed",false);CauldronManager.Instance.StopTasting();},3,null,1);Inv("Start");return "Queued equipment events and isolated autosave coordinator; private root checked and feedback submissions blocked.";}
}
