using System;using System.IO;using System.Collections;using System.Linq;using System.Linq.Expressions;using System.Reflection;using System.Collections.Generic;using UnityEngine;using UnityEditor;using TimelessEchoes;using TimelessEchoes.MapGeneration;using TimelessEchoes.Hero;using Blindsided.Utilities.Pooling;
public static class MapAuditObserver {
 const BindingFlags F=BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Static|BindingFlags.Instance;static object s;static Func<int> index;static Func<SegmentedMapGenerator,int> nextX;static int qi=-1,last=-1;static string scenario;static GameObject map;static SegmentedMapGenerator gen;static List<string> lines=new List<string>{"scenario,time,nextSegmentX,heroX,heroActive,generationComplete"};static List<string> pools=new List<string>{"scenario,pools,total,active,inactive"};
 static void Tick(){int i=index();if(i!=qi){qi=i;var sc=s.GetType().GetField("current",F).GetValue(s);scenario=(string)sc.GetType().GetField("name").GetValue(sc);int p=0,total=0,active=0,inactive=0;foreach(string field in new[]{"poolsByPrefabId","poolsByName"}){var d=(IDictionary)typeof(PoolManager).GetField(field,F).GetValue(null);foreach(DictionaryEntry e in d){var t=e.Value.GetType();p++;total+=(int)t.GetField("total",F).GetValue(e.Value);active+=(int)t.GetField("active",F).GetValue(e.Value);inactive+=(int)t.GetField("inactive",F).GetValue(e.Value);}}pools.Add($"{scenario},{p},{total},{active},{inactive}");}
 var m=GameManager.Instance.CurrentMap;if(m!=map){map=m;gen=m?m.GetComponentInChildren<SegmentedMapGenerator>():null;last=-1;}if(gen==null)return;int x=nextX(gen);if(x==last)return;last=x;var h=HeroController.Instance;lines.Add($"{scenario},{Time.realtimeSinceStartupAsDouble:R},{x},{(h?h.transform.position.x:0):R},{(h&&h.gameObject.activeInHierarchy)},{gen.InitialGenerationComplete}");}
 static void Stop(){EditorApplication.update-=Tick;File.WriteAllLines("docs/profiling/2026-09-26/chunk-observations.csv",lines);File.WriteAllLines("docs/profiling/2026-09-26/pool-observations.csv",pools);}
 public static object Main(){s=AppDomain.CurrentDomain.GetData("EOV_AUDIT");var st=s.GetType();index=Expression.Lambda<Func<int>>(Expression.Field(Expression.Convert(Expression.Constant(s),st),st.GetField("qi",F))).Compile();var p=Expression.Parameter(typeof(SegmentedMapGenerator));nextX=Expression.Lambda<Func<SegmentedMapGenerator,int>>(Expression.Field(p,typeof(SegmentedMapGenerator).GetField("nextSegmentX",F)),p).Compile();EditorApplication.update+=Tick;AppDomain.CurrentDomain.SetData("EOV_MAP_OBSERVER_CLEANUP",(Action)Stop);return "Allocation-free frame polling of chunk sequence; per-boundary pool counts; events emitted only on segment creation";}
}
