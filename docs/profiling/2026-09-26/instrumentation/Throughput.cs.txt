using System;using System.IO;using System.Linq;using System.Linq.Expressions;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEditor;using TimelessEchoes.Stats;using TimelessEchoes.Hero;
public static class ThroughputAudit {
 static object s;static Func<int> index;static Func<string> phase;static int prev=-999,frame=-1;static Observation item;static List<Observation> result=new List<Observation>();static double time;static int tasks,kills;static float distance;
 public class Observation{public int index;public string name;public double start,end;public int tasksStart,tasksEnd,killsStart,killsEnd;public float distanceStart,distanceEnd;public int echoesStart,echoesEnd;}
 static T Getter<T>(string name){var t=s.GetType();return Expression.Lambda<T>(Expression.Field(Expression.Convert(Expression.Constant(s),t),t.GetField(name,BindingFlags.Instance|BindingFlags.NonPublic)),Array.Empty<ParameterExpression>()).Compile();}
 static void Tick(){if(Time.frameCount==frame)return;frame=Time.frameCount;int now=index();string p=phase();if(now!=prev){if(item!=null){item.end=time;item.tasksEnd=tasks;item.killsEnd=kills;item.distanceEnd=distance;item.echoesEnd=UnityEngine.Object.FindObjectsByType<EchoController>(FindObjectsInactive.Exclude).Length;result.Add(item);item=null;}prev=now;}if(p!="capturing")return;var tr=GameplayStatTracker.Instance;if(tr==null)return;if(item==null){var current=s.GetType().GetField("current",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(s);item=new Observation{index=now,name=(string)current.GetType().GetField("name").GetValue(current),start=Time.realtimeSinceStartupAsDouble,tasksStart=tr.TasksCompleted,killsStart=tr.TotalKills,distanceStart=tr.CurrentRunDistance,echoesStart=UnityEngine.Object.FindObjectsByType<EchoController>(FindObjectsInactive.Exclude).Length};}time=Time.realtimeSinceStartupAsDouble;tasks=tr.TasksCompleted;kills=tr.TotalKills;distance=tr.CurrentRunDistance;}
 static void Stop(){EditorApplication.update-=Tick;if(item!=null){item.end=time;item.tasksEnd=tasks;item.killsEnd=kills;item.distanceEnd=distance;result.Add(item);item=null;}var t=AppDomain.CurrentDomain.GetAssemblies().First(a=>a.GetName().Name=="Newtonsoft.Json").GetType("Newtonsoft.Json.JsonConvert");File.WriteAllText("docs/profiling/2026-09-26/throughput-observations.json",(string)t.GetMethod("SerializeObject",new[]{typeof(object)}).Invoke(null,new object[]{result}));}
 public static object Main(){s=AppDomain.CurrentDomain.GetData("EOV_AUDIT");index=Getter<Func<int>>("qi");phase=Getter<Func<string>>("phase");EditorApplication.update+=Tick;AppDomain.CurrentDomain.SetData("EOV_THROUGHPUT_CLEANUP",(Action)Stop);return "Live tracker counters sampled without serializing save state; retained until cleanup";}
}
