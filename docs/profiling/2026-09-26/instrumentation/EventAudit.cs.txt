using System;using System.IO;using System.Linq;using System.Reflection;using System.Collections;using System.Collections.Generic;using UnityEngine;using TimelessEchoes.Upgrades;using Blindsided;
public static class EventAudit {
 const BindingFlags F=BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Instance|BindingFlags.Static;static object s;static List<Entry> entries=new List<Entry>();static Dictionary<FieldInfo,Delegate> original=new Dictionary<FieldInfo,Delegate>();static Dictionary<FieldInfo,object> owners=new Dictionary<FieldInfo,object>();static double began;static string name;
 public class Entry{public string source,method;public long count,ticks,maxTicks;public void Hit(Action a){long t=System.Diagnostics.Stopwatch.GetTimestamp();try{a();}finally{long dt=System.Diagnostics.Stopwatch.GetTimestamp()-t;count++;ticks+=dt;maxTicks=Math.Max(maxTicks,dt);}}}
 static object Inv(string n,params object[] a)=>s.GetType().GetMethod(n).Invoke(s,a);
 static void Wrap(object o,string n){var f=o.GetType().GetField(n,F);if(f==null)return;var d=f.GetValue(o) as Delegate;if(d==null)return;original[f]=d;owners[f]=o;Delegate all=null;foreach(var handler in d.GetInvocationList()){var e=new Entry{source=n,method=handler.Method.DeclaringType.FullName+"."+handler.Method.Name};entries.Add(e);Delegate wrapped=null;if(handler is Action a)wrapped=(Action)(()=>e.Hit(a));else if(handler is Action<int> ai)wrapped=(Action<int>)(x=>e.Hit(()=>ai(x)));else if(handler is Action<string,int> asi)wrapped=(Action<string,int>)((x,y)=>e.Hit(()=>asi(x,y)));else if(handler is Action<CauldronManager.TastingStats> at)wrapped=(Action<CauldronManager.TastingStats>)(x=>e.Hit(()=>at(x)));else continue;all=Delegate.Combine(all,wrapped);}if(all!=null)f.SetValue(o,all);}
 static void Unwrap(){foreach(var kv in original)kv.Key.SetValue(owners[kv.Key],kv.Value);original.Clear();owners.Clear();}
 static void Save(){if(name==null)return;Unwrap();double factor=1000.0/System.Diagnostics.Stopwatch.Frequency;var lines=new List<string>{"event,handler,calls,total_ms,max_ms,elapsed_s"};foreach(var e in entries)lines.Add($"{e.source},{e.method},{e.count},{e.ticks*factor:R},{e.maxTicks*factor:R},{Time.realtimeSinceStartupAsDouble-began:R}");File.WriteAllLines("docs/profiling/2026-09-26/"+name+"-handlers.csv",lines);name=null;}
 public static object Main(){s=AppDomain.CurrentDomain.GetData("EOV_AUDIT");AppDomain.CurrentDomain.SetData("EOV_EVENT_CLEANUP",(Action)Save);var q=(IList)s.GetType().GetField("queue").GetValue(s);int old=q.Count;
 foreach(bool open in new[]{false,true}){bool v=open;Inv("Add","event-taste-100-"+(v?"open":"closed"),(Action)(()=>{Save();Inv("Reset","developed",false);if(v)Inv("Panel","cauldron");var cm=CauldronManager.Instance;Oracle.oracle.saveData.CauldronStew=1e9;cm.StartTasting();entries.Clear();name="event-taste-100-"+(v?"open":"closed");began=Time.realtimeSinceStartupAsDouble;foreach(var n in new[]{"OnStewChanged","OnWeightsChanged","OnCardGained","OnSessionCardsChanged","OnStatsChanged"})Wrap(cm,n);Wrap(ResourceManager.Instance,"OnInventoryChanged");}),20.0,null,5.0);}
 Inv("Add","event-cleanup",(Action)(()=>{Save();Inv("Reset","developed",false);CauldronManager.Instance.StopTasting();}),2.0,null,1.0);
 for(int rep=1;rep<=3;rep++)Inv("Add","town-developed-no-taste-r"+rep,(Action)(()=>{Inv("Reset","developed",false);CauldronManager.Instance.StopTasting();}),15.0,null,5.0);
 object[] added=new object[q.Count-old];for(int i=0;i<added.Length;i++)added[i]=q[old+i];for(int i=q.Count-1;i>=old;i--)q.RemoveAt(i);int at=34;foreach(var a in added)q.Insert(at++,a);return "Targeted event timings and no-tasting town controls queued";
 }
}
