using System;using System.IO;using System.Linq;using System.Reflection;using System.Collections;using UnityEngine;using Blindsided;using Blindsided.SaveData;using TimelessEchoes.Gear;using TimelessEchoes.Gear.UI;using TimelessEchoes.Upgrades;using TimelessEchoes.UI;
public static class ForgeAudit {
 const BindingFlags F=BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Instance|BindingFlags.Static;static object s;static ForgeWindowUI f;static CraftingConfigSO orig,clone;static Resource[] resources;static CoreSO core;
 static object Inv(string n,params object[] a)=>s.GetType().GetMethod(n).Invoke(s,a);static object Get(object o,string n)=>o.GetType().GetField(n,F).GetValue(o);static void Set(object o,string n,object v)=>o.GetType().GetField(n,F).SetValue(o,v);static object Call(object o,string n,params object[] a)=>o.GetType().GetMethods(F).First(m=>m.Name==n&&m.GetParameters().Length==a.Length).Invoke(o,a);
 static void Add(string n,Action a,double sec=15,Action tick=null,double warm=5)=>Inv("Add",n,a,sec,tick,warm);
 static void Prepare(string name,bool early=false){Inv("Reset",early?"fresh":"developed",false);Inv("Panel","forge");core=Resources.Load<CoreSO>("Gear/Cores/"+name);foreach(var r in resources)Oracle.oracle.saveData.Resources[r.name]=new GameData.ResourceEntry{Amount=1e8,Earned=true,Tier=1};Blindsided.EventHandler.LoadData();Call(f,"SelectCore",core);f.SelectSlot("Weapon");}
 public static object Main(){s=AppDomain.CurrentDomain.GetData("EOV_AUDIT");var q=(IList)s.GetType().GetField("queue").GetValue(s);int old=q.Count;f=UnityEngine.Object.FindObjectsByType<ForgeWindowUI>(FindObjectsInactive.Include).First();resources=Resources.LoadAll<Resource>("Resource Items");orig=CraftingService.Instance.Config;clone=UnityEngine.Object.Instantiate(orig);clone.hideFlags=HideFlags.HideAndDontSave;AppDomain.CurrentDomain.SetData("EOV_CRAFT_ORIGINAL",orig);AppDomain.CurrentDomain.SetData("EOV_CRAFT_CLONE",clone);
 Add("forge-manual-early",()=>Prepare("Eznorb",true),20,()=>Call(f,"OnCraftClicked"));
 foreach(var tier in new[]{"Eznorb","Dlog","Vastium"}){var t=tier;Add("forge-manual-developed-"+tier,()=>Prepare(t),20,()=>Call(f,"OnCraftClicked"));}
 foreach(int rate in new[]{20,100,1000})for(int rep=1;rep<=3;rep++){int r=rate;Add("forge-auto-config-"+rate+"-r"+rep,()=>{Prepare("Eznorb");clone.baseCraftsPerSecond=r;clone.maxCraftsPerSecond=r;Set(CraftingService.Instance,"config",clone);Call(f,"OnCraftUntilUpgradeClicked");},20);}
 Add("forge-close-stops-auto",()=>{Prepare("Eznorb");Call(f,"OnCraftUntilUpgradeClicked");TownWindowManager.Instance.CloseAllWindows();},15);
 Add("forge-behind-cauldron",()=>{Prepare("Eznorb");Call(f,"OnCraftUntilUpgradeClicked");var refPanel=Get(TownWindowManager.Instance,"cauldron");((GameObject)Get(refPanel,"window")).SetActive(true);Oracle.oracle.saveData.CauldronStew=1e9;CauldronManager.Instance.StartTasting();},30);
 Add("forge-restore-config",()=>{Inv("Reset","developed",false);Set(CraftingService.Instance,"config",orig);},2,null,1);
 foreach(int cap in new[]{30,60,144})for(int rep=1;rep<=3;rep++){int c=cap;Add("taste-native-cap"+cap+"-r"+rep,()=>{Inv("Reset","developed",false);Application.targetFrameRate=c;QualitySettings.vSyncCount=0;Oracle.oracle.saveData.CauldronStew=1e9;CauldronManager.Instance.StartTasting();},20);}
 Add("restore-uncapped",()=>Application.targetFrameRate=-1,2,null,1);
 foreach(var stage in new[]{"empty-cards","tier-boundary","maxed-infinity"})foreach(bool visible in new[]{false,true}){string st=stage;bool v=visible;Add("taste-"+st+"-"+(v?"open":"closed"),()=>{Inv("Reset","developed",false);var d=Oracle.oracle.saveData;if(st=="empty-cards"){d.CauldronCardCounts.Clear();d.CauldronEvaLevel=1;d.CauldronEvaXp=0;}else{foreach(var key in d.CauldronCardCounts.Keys.ToArray())if(!key.StartsWith("INF:"))d.CauldronCardCounts[key]=st=="tier-boundary"?(key.StartsWith("RES:")?9999:2999):(key.StartsWith("RES:")?10000:3000);}Blindsided.EventHandler.LoadData();d.CauldronStew=1e9;if(v)Inv("Panel","cauldron");CauldronManager.Instance.StartTasting();},20);}
 int cycle=0;foreach(int round in new[]{1,2,3}){Add("ui-cycles-"+round,()=>{Inv("Reset","developed",false);cycle=0;},45,()=>{Inv("Panel",new[]{"forge","cauldron","quests","skills","wiki","inventory","stats",null}[cycle++%8]);});Add("ui-retained-"+round,()=>{Inv("Panel",new object[]{null});GC.Collect();GC.WaitForPendingFinalizers();},10);}
 var added=new object[q.Count-old];for(int i=0;i<added.Length;i++)added[i]=q[old+i];for(int i=q.Count-1;i>=old;i--)q.RemoveAt(i);int at=34;foreach(var a in added)q.Insert(at++,a);
 return new{added=added.Length,total=q.Count};}
}
