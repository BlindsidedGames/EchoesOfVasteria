using System;using System.IO;using System.Linq;using System.Linq.Expressions;using System.Reflection;using System.Collections.Generic;using UnityEngine;using UnityEditor;using TimelessEchoes.Gear;
public static class CraftObserver {
 static object s;static Func<int> index;static Func<string> phase;static int prev=-1,count,startCount,lastCount;static double start,last;static string name;static List<string> lines=new List<string>{"scenario,crafts,seconds,crafts_per_s"};static Action<int,float,float> handler=(a,b,c)=>count++;
 static void Tick(){int i=index();if(i!=prev){if(name!=null){double dt=last-start;lines.Add($"{name},{lastCount-startCount},{dt:R},{(lastCount-startCount)/dt:R}");name=null;}prev=i;}if(phase()!="capturing")return;if(name==null){var c=s.GetType().GetField("current",BindingFlags.NonPublic|BindingFlags.Instance).GetValue(s);name=(string)c.GetType().GetField("name").GetValue(c);start=Time.realtimeSinceStartupAsDouble;startCount=count;}last=Time.realtimeSinceStartupAsDouble;lastCount=count;}
 static void Stop(){EditorApplication.update-=Tick;CraftingService.Instance.OnIvanXpChanged-=handler;File.WriteAllLines("docs/profiling/2026-09-26/craft-throughput.csv",lines);}
 public static object Main(){s=AppDomain.CurrentDomain.GetData("EOV_AUDIT");var t=s.GetType();var inst=Expression.Convert(Expression.Constant(s),t);index=Expression.Lambda<Func<int>>(Expression.Field(inst,t.GetField("qi",BindingFlags.NonPublic|BindingFlags.Instance))).Compile();phase=Expression.Lambda<Func<string>>(Expression.Field(inst,t.GetField("phase",BindingFlags.NonPublic|BindingFlags.Instance))).Compile();CraftingService.Instance.OnIvanXpChanged+=handler;EditorApplication.update+=Tick;AppDomain.CurrentDomain.SetData("EOV_CRAFT_OBSERVER_CLEANUP",(Action)Stop);return "Successful craft XP event counter attached; first partial scenario must be excluded";}
}
