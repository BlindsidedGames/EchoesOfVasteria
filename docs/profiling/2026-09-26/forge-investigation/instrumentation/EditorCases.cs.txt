using System;using System.IO;using System.Linq;using System.Reflection;using System.Collections.Generic;using UnityEngine;using UnityEngine.UI;using UnityEditor;using UnityEditorInternal;using Blindsided;using Blindsided.SaveData;using TimelessEchoes;using TimelessEchoes.UI;using TimelessEchoes.Upgrades;using TimelessEchoes.Gear;using TimelessEchoes.Gear.UI;
public static class ForgeEditorCases {
 const BindingFlags F=BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Instance|BindingFlags.Static;static object s;static ForgeWindowUI forge;static CoreSO core;static Resource resource;static int cycles;static List<Behaviour> frozen=new List<Behaviour>();static string dir="docs/profiling/2026-09-26/forge-investigation/editor/";static EditorWindow profiler;static bool profilerBefore;static int first;
 static object Get(object o,string n)=>o.GetType().GetField(n,F).GetValue(o);static object Call(object o,string n,params object[] a)=>o.GetType().GetMethods(F).First(m=>m.Name==n&&m.GetParameters().Length==a.Length).Invoke(o,a);static object Inv(string n,params object[] a)=>Call(s,n,a);static void Add(string n,Action a,double sec=15,Action tick=null,double warm=5)=>Inv("Add",n,a,sec,tick,warm);
 static void Prepare(bool visible,bool freeze,bool select,bool taste){foreach(var b in frozen)if(b!=null)b.enabled=true;frozen.Clear();Inv("Reset","developed",false);foreach(var r in Resources.LoadAll<Resource>("Resource Items"))Oracle.oracle.saveData.Resources[r.name]=new GameData.ResourceEntry{Amount=1e8,Earned=true,Tier=1};Blindsided.EventHandler.LoadData();Call(forge,"StopAutoCrafting");if(!taste)CauldronManager.Instance.StopTasting();if(select){Call(forge,"SelectCore",core);forge.SelectSlot("Weapon");}Inv("Panel",visible?"forge":null);QualitySettings.vSyncCount=select?0:1;Application.targetFrameRate=select?120:-1;cycles=0;if(freeze){Canvas.ForceUpdateCanvases();var w=(GameObject)Get(Get(TownWindowManager.Instance,"forge"),"window");foreach(var b in w.GetComponentsInChildren<Behaviour>(true))if(b.enabled&&(b is LayoutGroup||b is ContentSizeFitter)){frozen.Add(b);b.enabled=false;}}}
 static void Cycle(){string slot=new[]{"Weapon","Chest","Helmet","Boots"}[cycles++%4];var item=CraftingService.Instance.Craft(core,slot,null,resource,1);if(item==null)throw new Exception("Craft failed");var old=EquipmentController.Instance.GetEquipped(slot);EquipmentController.Instance.Equip(item);if(old!=null)SalvageService.Instance.Salvage(old,true);}
 public class Metric{public string name;public long count;public double inclusive,self,max;}
 static void EndTrace(){if(profiler==null)return;ProfilerDriver.enabled=false;var data=new Dictionary<string,Metric>();int valid=0;for(int f=Math.Max(first,ProfilerDriver.firstFrameIndex);f<=ProfilerDriver.lastFrameIndex;f++){using(var v=ProfilerDriver.GetRawFrameDataView(f,0)){if(!v.valid)continue;valid++;for(int i=0;i<v.sampleCount;i++){var n=v.GetSampleName(i);double ms=v.GetSampleTimeMs(i),child=0;int j=i+1;for(int c=0;c<v.GetSampleChildrenCount(i)&&j<v.sampleCount;c++){child+=v.GetSampleTimeMs(j);j+=v.GetSampleChildrenCountRecursive(j)+1;}if(!data.TryGetValue(n,out var m)){m=new Metric{name=n};data[n]=m;}m.count++;m.inclusive+=ms;m.self+=Math.Max(0,ms-child);m.max=Math.Max(m.max,ms);}}}using(var w=new StreamWriter(dir+"forge-timeline.csv")){w.WriteLine("name,calls,inclusive_ms,self_ms,max_ms,frames");foreach(var m in data.Values.OrderByDescending(m=>m.self))w.WriteLine("\""+m.name.Replace("\"","\"\"")+"\","+m.count+","+m.inclusive+","+m.self+","+m.max+","+valid);}ProfilerDriver.enabled=profilerBefore;profiler.Close();profiler=null;}
 public static object Main(){s=AppDomain.CurrentDomain.GetData("EOV_AUDIT");forge=UnityEngine.Object.FindObjectsByType<ForgeWindowUI>(FindObjectsInactive.Include).First();core=Resources.Load<CoreSO>("Gear/Cores/Vastium");resource=Resources.LoadAll<Resource>("Resource Items").First(r=>r.name=="Vastium Core");
 for(int r=1;r<=2;r++){Add("editor-original-setup-open-r"+r,()=>Prepare(true,false,false,true),15,Cycle);Add("editor-matched-player-open-r"+r,()=>Prepare(true,false,true,false),15,Cycle);Add("editor-layout-frozen-r"+r,()=>Prepare(true,true,true,false),15,Cycle);Add("editor-closed-r"+r,()=>Prepare(false,false,true,false),15,Cycle);}
 Add("editor-timeline",()=>{Prepare(true,false,false,true);profilerBefore=ProfilerDriver.enabled;profiler=EditorWindow.GetWindow(typeof(EditorWindow).Assembly.GetType("UnityEditor.ProfilerWindow"));first=ProfilerDriver.lastFrameIndex+1;ProfilerDriver.enabled=true;},5,Cycle,3);
 Add("editor-cleanup",()=>{EndTrace();Prepare(false,false,false,false);},3,null,1);AppDomain.CurrentDomain.SetData("FORGE_PROBE_EDITOR_CLEANUP",(Action)(()=>{EndTrace();foreach(var b in frozen)if(b!=null)b.enabled=true;frozen.Clear();}));Inv("Start");return "Queued controlled Editor forge repetitions and targeted timeline";}
}
